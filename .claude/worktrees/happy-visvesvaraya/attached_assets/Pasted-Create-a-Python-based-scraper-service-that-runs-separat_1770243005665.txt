EXAMPLE SCRAPER IMPLEMENTATION:
```python
# scrapers/sources/generic_scraper.py
from playwright.sync_api import sync_playwright
from bs4 import BeautifulSoup
import requests
from utils.db import upsert_grant, log_scrape_result

class GenericScraper:
    def __init__(self, source):
        self.source = source
        
    def scrape(self):
        try:
            if self.source['scraper_type'] == 'playwright':
                grants = self.scrape_with_playwright()
            elif self.source['scraper_type'] == 'beautifulsoup':
                grants = self.scrape_with_beautifulsoup()
            elif self.source['scraper_type'] == 'api':
                grants = self.scrape_api()
            
            # Upsert grants to DB
            for grant in grants:
                upsert_grant(grant)
            
            log_scrape_result(
                source_id=self.source['id'],
                status='success',
                grants_found=len(grants)
            )
            
        except Exception as e:
            log_scrape_result(
                source_id=self.source['id'],
                status='failed',
                error_message=str(e)
            )
    
    def scrape_with_playwright(self):
        with sync_playwright() as p:
            browser = p.chromium.launch()
            page = browser.new_page()
            page.goto(self.source['url'])
            
            # Wait for content to load
            page.wait_for_selector(self.source['selectors']['title_selector'])
            
            # Extract data
            html = page.content()
            grants = self.extract_grants(html)
            
            browser.close()
            return grants
    
    def extract_grants(self, html):
        soup = BeautifulSoup(html, 'html.parser')
        grants = []
        
        # Use selectors from DB config
        # This is simplified - you'll need to handle multiple items
        title = soup.select_one(self.source['selectors']['title_selector'])
        
        grant = {
            'title': title.text.strip() if title else None,
            'source_name': self.source['name'],
            'url': self.source['url'],
            # ... more fields
        }
        
        grants.append(grant)
        return grants
```

API ENDPOINT TO TRIGGER SCRAPER:

app/api/scraper/run/route.ts:
```typescript
import { exec } from 'child_process';

export async function POST(request: Request) {
  const { sourceId } = await request.json();
  
  // Trigger Python scraper
  exec(`python3 scrapers/main.py --source-id ${sourceId}`, (error, stdout, stderr) => {
    if (error) {
      console.error(`Error: ${error}`);
      return;
    }
    console.log(`Output: ${stdout}`);
  });
  
  return Response.json({ message: 'Scraper started' });
}
```
