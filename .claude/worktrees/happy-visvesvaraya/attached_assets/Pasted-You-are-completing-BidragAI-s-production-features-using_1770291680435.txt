STEP 1.3: Create Auth Middleware

Create server/middleware/auth.ts:
```typescript
import { Request, Response, NextFunction } from 'express';

// Extend Express Request type to include user
declare global {
  namespace Express {
    interface Request {
      user?: {
        id: string;
        replitUserId: string;
        email: string;
        fullName?: string;
      };
    }
  }
}

export async function requireAuth(req: Request, res: Response, next: NextFunction) {
  // Replit Auth provides user info via headers
  const replitUserId = req.headers['x-replit-user-id'] as string;
  const replitUserName = req.headers['x-replit-user-name'] as string;
  const replitUserEmail = req.headers['x-replit-user-email'] as string;

  if (!replitUserId) {
    return res.status(401).json({
      error: 'Unauthorized',
      message: 'Du måste vara inloggad för att komma åt denna resurs',
    });
  }

  // Find or create user in database
  const db = req.app.get('db'); // Assuming you store db instance in app
  
  let user = await db.query.users.findFirst({
    where: eq(users.replitUserId, replitUserId),
  });

  if (!user) {
    // Create user on first login
    [user] = await db.insert(users).values({
      replitUserId,
      email: replitUserEmail || `${replitUserName}@replit.com`,
      fullName: replitUserName,
    }).returning();
  }

  // Attach user to request
  req.user = {
    id: user.id,
    replitUserId: user.replitUserId,
    email: user.email,
    fullName: user.fullName,
  };

  next();
}

export function optionalAuth(req: Request, res: Response, next: NextFunction) {
  const replitUserId = req.headers['x-replit-user-id'] as string;
  
  if (replitUserId) {
    // Run requireAuth logic but don't fail if no user
    requireAuth(req, res, next);
  } else {
    next();
  }
}
```

STEP 3.3: Create Notification Cron Endpoint

Create server/routes/cron.ts:
```typescript
import express from 'express';
import { sendNewGrantEmail, sendDeadlineReminder } from '../services/email';

const router = express.Router();

// Middleware to verify cron secret
function verifyCronSecret(req, res, next) {
  const secret = req.headers['authorization']?.replace('Bearer ', '');
  
  if (secret !== process.env.CRON_SECRET) {
    return res.status(401).json({ error: 'Unauthorized' });
  }
  
  next();
}

router.use(verifyCronSecret);

// Daily notifications check
router.get('/cron/notifications', async (req, res) => {
  const db = req.app.get('db');
  
  try {
    // Get grants created in last 24 hours
    const yesterday = new Date(Date.now() - 24 * 60 * 60 * 1000);
    
    const newGrants = await db.query.grants.findMany({
      where: and(
        gte(grants.createdAt, yesterday),
        eq(grants.status, 'open')
      ),
    });
    
    // Get all users with email notifications enabled
    const usersWithProfiles = await db.query.users.findMany({
      with: {
        profile: true,
        companies: true,
      },
    });
    
    let sentCount = 0;
    
    for (const grant of newGrants) {
      for (const user of usersWithProfiles) {
        // Skip if notifications disabled
        if (!user.profile?.emailNotifications) continue;
        
        for (const company of user.companies) {
          // Calculate basic match
          const matchScore = calculateMatchScore(company, grant);
          
          if (matchScore >= 60) {
            await sendNewGrantEmail(
              user.email,
              grant,
              company.companyName,
              matchScore
            );
            
            // Log notification
            await db.insert(notifications).values({
              userId: user.id,
              grantId: grant.id,
              type: 'new_grant_match',
            });
            
            sentCount++;
          }
        }
      }
    }
    
    // Check upcoming deadlines (7, 3, 1 days)
    const today = new Date();
    const deadlineChecks = [7, 3, 1];
    
    for (const days of deadlineChecks) {
      const targetDate = new Date(today);
      targetDate.setDate(targetDate.getDate() + days);
      targetDate.setHours(0, 0, 0, 0);
      
      const endOfDay = new Date(targetDate);
      endOfDay.setHours(23, 59, 59, 999);
      
      const upcomingGrants = await db.query.grants.findMany({
        where: and(
          eq(grants.status, 'open'),
          gte(grants.deadline, targetDate),
          lte(grants.deadline, endOfDay)
        ),
      });
      
      for (const grant of upcomingGrants) {
        for (const user of usersWithProfiles) {
          if (!user.profile?.emailNotifications) continue;
          
          // Check if already notified in last 24h
          const recentNotif = await db.query.notifications.findFirst({
            where: and(
              eq(notifications.userId, user.id),
              eq(notifications.grantId, grant.id),
              eq(notifications.type, 'deadline_reminder'),
              gte(notifications.sentAt, yesterday)
            ),
          });
          
          if (!recentNotif && user.companies.length > 0) {
            await sendDeadlineReminder(
              user.email,
              grant,
              user.companies[0].companyName,
              days
            );
            
            await db.insert(notifications).values({
              userId: user.id,
              grantId: grant.id,
              type: 'deadline_reminder',
            });
            
            sentCount++;
          }
        }
      }
    }
    
    res.json({
      success: true,
      notificationsSent: sentCount,
      newGrants: newGrants.length,
    });
    
  } catch (error) {
    console.error('Notification cron failed:', error);
    res.status(500).json({
      success: false,
      error: error.message,
    });
  }
});

export default router;

function calculateMatchScore(company, grant) {
  // Simplified matching logic
  let score = 50;
  
  if (grant.keywords?.some(k => 
    company.description?.toLowerCase().includes(k.toLowerCase())
  )) {
    score += 20;
  }
  
  if (grant.targetGroup?.includes(company.industry)) {
    score += 30;
  }
  
  return Math.min(100, score);
}
```
