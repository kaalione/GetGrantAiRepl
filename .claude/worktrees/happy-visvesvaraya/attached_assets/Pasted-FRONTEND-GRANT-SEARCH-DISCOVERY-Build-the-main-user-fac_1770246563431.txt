MATCHING LOGIC (basic):

Create lib/matching.ts:
```typescript
export function calculateMatchScore(company, grant) {
  let score = 0;
  let maxScore = 100;
  
  // Industry match (30 points)
  if (grant.target_group?.includes(company.industry)) {
    score += 30;
  }
  
  // Size match (20 points)
  if (company.employees >= grant.eligibility_criteria?.min_employees &&
      company.employees <= grant.eligibility_criteria?.max_employees) {
    score += 20;
  }
  
  // Revenue match (20 points)
  if (company.revenue >= grant.eligibility_criteria?.min_revenue &&
      company.revenue <= grant.eligibility_criteria?.max_revenue) {
    score += 20;
  }
  
  // Location match (15 points)
  if (grant.eligibility_criteria?.regions?.includes(company.location)) {
    score += 15;
  }
  
  // Keywords overlap (15 points)
  const companyKeywords = company.description.toLowerCase().split(' ');
  const matchedKeywords = grant.keywords?.filter(k => 
    companyKeywords.includes(k.toLowerCase())
  ).length || 0;
  score += Math.min(15, matchedKeywords * 3);
  
  return Math.min(score, maxScore);
}
```
