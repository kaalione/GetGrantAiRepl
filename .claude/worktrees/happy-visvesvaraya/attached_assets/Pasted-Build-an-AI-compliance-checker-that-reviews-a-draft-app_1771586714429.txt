Build an AI compliance checker that reviews a draft application against the 
grant's official evaluation criteria and identifies gaps, weaknesses, and 
missing requirements before submission.

## Overview
After a user has written or AI-generated their application, they can run a 
compliance check. The AI reviews each section against what evaluators actually 
look for and returns specific, actionable feedback — not just "good" or "bad" 
but exactly what is missing and how to fix it.

Inspired by Cogrant Write's compliance checks and Granter.ai's criteria alignment.

## Backend: server/services/complianceChecker.ts
```typescript
import Anthropic from '@anthropic-ai/sdk';

interface ComplianceReport {
  overallScore: number;             // 0-100 readiness score
  readinessLevel: 'not_ready' | 'needs_work' | 'almost_ready' | 'ready';
  sections: SectionCompliance[];
  criticalIssues: string[];         // blockers that will likely cause rejection
  improvements: string[];           // high-impact improvements
  strengths: string[];              // what's already working well
  estimatedSuccessRate: string;     // e.g. "Low (20-30%)" | "Medium (40-60%)" | "High (70%+)"
  checkedAt: string;
}

interface SectionCompliance {
  sectionKey: string;
  sectionTitle: string;
  score: number;                    // 0-100
  issues: ComplianceIssue[];
  suggestions: string[];
}

interface ComplianceIssue {
  severity: 'critical' | 'major' | 'minor';
  issue: string;                    // what's wrong
  criterion: string;                // which evaluation criterion is affected
  fix: string;                      // specific actionable fix
}

export async function checkCompliance(
  grant: Grant,
  application: Application,
  companyProfile: CompanyProfile
): Promise<ComplianceReport> {
  const client = new Anthropic();
  
  // Build the full application text for review
  const applicationText = application.sections
    .map(s => `## ${s.sectionTitle}\n${s.content}`)
    .join('\n\n');
  
  // Get grant-specific evaluation criteria
  const evaluationCriteria = getEvaluationCriteria(grant.source);
  
  const response = await client.messages.create({
    model: 'claude-opus-4-5',
    max_tokens: 2000,
    system: `You are a senior grant evaluator with 15+ years experience reviewing 
applications for ${grant.source}. You know exactly what makes applications succeed 
or fail. Be specific, honest, and constructive. Always respond with valid JSON only.`,
    messages: [{
      role: 'user',
      content: buildCompliancePrompt(grant, applicationText, evaluationCriteria, companyProfile)
    }]
  });
  
  const text = response.content[0].type === 'text' ? response.content[0].text : '{}';
  
  try {
    const result = JSON.parse(text) as ComplianceReport;
    result.checkedAt = new Date().toISOString();
    return result;
  } catch {
    return buildFallbackReport();
  }
}

function getEvaluationCriteria(source: string): string {
  const criteria: Record<string, string> = {
    vinnova: `
      Vinnova evaluates on 4 criteria (equal weight):
      1. RELEVANCE: Does the project address important challenges? Is there a clear 
         societal or economic need? Is the timing right?
      2. SCIENTIFIC/TECHNICAL QUALITY: Is the approach innovative? Is the methodology 
         sound? What is the current TRL and what TRL will be reached?
      3. FEASIBILITY: Is the team competent? Is the plan realistic? Are risks identified 
         and managed? Is the budget reasonable?
      4. IMPACT: What are the expected results? How will results be used and spread? 
         What is the commercialization plan? What is the potential for Swedish 
         competitiveness?
      
      Common rejection reasons: vague impact statements, budget not justified line-by-line,
      team lacks relevant competence, no evidence of market need, too early stage for 
      the call, collaboration requirements not met.`,
      
    tillvaxtverket: `
      Tillväxtverket evaluates on:
      1. NEED AND PROBLEM: Is there a documented market failure? Why can't the market 
         solve this alone?
      2. REGIONAL IMPACT: Does the project create jobs in the region? Does it strengthen 
         regional competitiveness? Is there a clear connection to regional strategy?
      3. ADDITIONALITY: Would the project happen without the grant? Can the company 
         co-finance (typically 50%+)?
      4. SUSTAINABILITY: Will results last beyond the project? Environmental impact?
      
      Common rejection reasons: insufficient regional connection, no co-financing plan,
      project could be funded commercially, unclear job creation numbers.`,
      
    energimyndigheten: `
      Energimyndigheten evaluates on:
      1. ENERGY RELEVANCE: Clear connection to Swedish energy transition goals
      2. INNOVATION LEVEL: Is this genuinely new? What is TRL? 
      3. COMMERCIAL POTENTIAL: Path to market within 5 years?
      4. TEAM: Energy sector competence, relevant partners
      5. ENVIRONMENTAL BENEFIT: Quantified CO2 reduction potential (tonnes/year)
      
      Common rejection reasons: no quantified energy savings, missing industry partner,
      TRL too high (basic research) or too low (already commercial).`,

    generic: `
      Standard grant evaluation criteria:
      1. Problem/need clearly defined with evidence
      2. Solution is innovative and feasible
      3. Team has relevant expertise and capacity
      4. Budget is realistic and well-justified
      5. Expected outcomes are measurable and significant
      6. Sustainability plan beyond the grant period
      7. Compliance with all eligibility requirements`
  };
  
  return criteria[source.toLowerCase()] || criteria.generic;
}

function buildCompliancePrompt(
  grant: Grant,
  applicationText: string,
  evaluationCriteria: string,
  company: CompanyProfile
): string {
  return `Review this grant application for compliance and quality.

GRANT:
Title: ${grant.title}
Funder: ${grant.source}
Description: ${grant.description?.substring(0, 400)}
Max amount: ${grant.maxAmount ? `${grant.maxAmount} SEK` : 'Not specified'}

OFFICIAL EVALUATION CRITERIA:
${evaluationCriteria}

APPLICANT COMPANY:
${company.name} | ${company.sector} | ${company.employees} employees | ${company.region}

APPLICATION TEXT:
${applicationText}

Review each section of the application against the evaluation criteria. 
Be specific — quote or reference actual text from the application when identifying issues.
Focus on what will actually affect the evaluation score, not stylistic preferences.

Respond with JSON only:
{
  "overallScore": 0-100,
  "readinessLevel": "not_ready" | "needs_work" | "almost_ready" | "ready",
  "estimatedSuccessRate": "Low (15-25%)" | "Medium (35-55%)" | "High (65-80%)",
  "criticalIssues": [
    "Specific critical issue that will cause rejection — quote from application if relevant"
  ],
  "improvements": [
    "High-impact improvement with specific action: what to add/change/remove"
  ],
  "strengths": [
    "What the application does well — be specific"
  ],
  "sections": [
    {
      "sectionKey": "section_key",
      "sectionTitle": "Section Title",
      "score": 0-100,
      "issues": [
        {
          "severity": "critical" | "major" | "minor",
          "issue": "What is wrong or missing",
          "criterion": "Which evaluation criterion this affects",
          "fix": "Specific action to fix this: add X, change Y to Z, include data on W"
        }
      ],
      "suggestions": [
        "Optional improvement that would strengthen this section"
      ]
    }
  ]
}`;
}
```

🟡 MEDIUM 3: Smart Content Library
Build a Smart Content Library that learns from a user's previous applications 
and company documents to build reusable content blocks — so users never write 
the same paragraph about their company twice.

## Overview
Every grant application requires the same building blocks: company description, 
team overview, problem statement, market analysis. Instead of rewriting these 
from scratch each time, the Content Library stores approved versions that the 
AI uses as foundation for new applications. Inspired by Grantable's "learns 
from previous applications" feature and OpenGrants' content repository.

## Database schema

Add to schema.ts:
```typescript
export const contentLibrary = pgTable('content_library', {
  id: uuid('id').primaryKey().defaultRandom(),
  userId: varchar('user_id').notNull(),
  companyId: varchar('company_id').notNull(),
  contentType: varchar('content_type').notNull(), 
  // Types: company_description | team_overview | problem_statement | 
  //        market_analysis | technology_description | impact_statement | 
  //        budget_approach | sustainability_plan | custom
  title: varchar('title').notNull(),              // user-given name for this block
  content: text('content').notNull(),
  wordCount: integer('word_count'),
  language: varchar('language').default('sv'),    // sv | en
  tags: varchar('tags').array().default('[]'),    // e.g. ['vinnova', 'technical', '2025']
  usageCount: integer('usage_count').default(0),  // how many times used in applications
  lastUsedAt: timestamp('last_used_at'),
  sourceApplicationId: uuid('source_application_id'), // if extracted from past application
  isApproved: boolean('is_approved').default(false),  // user has reviewed and approved
  createdAt: timestamp('created_at').defaultNow(),
  updatedAt: timestamp('updated_at').defaultNow(),
});

// Track which content blocks were used in which applications
export const contentLibraryUsage = pgTable('content_library_usage', {
  id: uuid('id').primaryKey().defaultRandom(),
  contentId: uuid('content_id').notNull().references(() => contentLibrary.id),
  applicationId: uuid('application_id').notNull().references(() => applications.id),
  sectionKey: varchar('section_key').notNull(),
  usedAt: timestamp('used_at').defaultNow(),
});
```

## Backend: server/services/contentLibrary.ts

### Auto-extraction from completed applications
```typescript
import Anthropic from '@anthropic-ai/sdk';

export async function extractContentBlocks(
  application: Application,
  companyProfile: CompanyProfile
): Promise<ExtractedContent[]> {
  const client = new Anthropic();
  
  const fullText = application.sections
    .map(s => `[${s.sectionKey}]\n${s.content}`)
    .join('\n\n');
  
  const response = await client.messages.create({
    model: 'claude-haiku-4-5',
    max_tokens: 2000,
    system: 'You extract reusable content blocks from grant applications. Respond with JSON only.',
    messages: [{
      role: 'user',
      content: `Extract reusable content blocks from this grant application.
For each block, extract content that could be reused in future applications 
with minimal editing. Focus on evergreen content about the company, team, 
technology, and problem — NOT grant-specific content like budget justifications.

APPLICATION:
${fullText}

COMPANY: ${companyProfile.name} | ${companyProfile.sector}

Extract blocks in JSON array:
[
  {
    "contentType": "company_description" | "team_overview" | "problem_statement" | 
                   "market_analysis" | "technology_description" | "impact_statement" | 
                   "sustainability_plan",
    "title": "Short descriptive name for this block",
    "content": "The extracted reusable text",
    "language": "sv" | "en",
    "tags": ["relevant", "tags"],
    "qualityNote": "Brief note on what makes this reusable / what would need editing"
  }
]

Only extract blocks that are:
1. At least 50 words long
2. About the company/project in general (not grant-specific)
3. Well-written enough to reuse
Return empty array if no good reusable content found.`
    }]
  });
  
  const text = response.content[0].type === 'text' ? response.content[0].text : '[]';
  try {
    return JSON.parse(text);
  } catch {
    return [];
  }
}

export async function findRelevantContentBlocks(
  sectionType: string,
  grant: Grant,
  companyId: string,
  language: string = 'sv'
): Promise<ContentBlock[]> {
  // Find content blocks that match the section type and language
  const blocks = await db.query.contentLibrary.findMany({
    where: and(
      eq(contentLibrary.companyId, companyId),
      eq(contentLibrary.isApproved, true),
      eq(contentLibrary.language, language)
    ),
    orderBy: [desc(contentLibrary.usageCount), desc(contentLibrary.updatedAt)],
    limit: 10
  });
  
  // Filter by relevance to section type
  const typeMapping: Record<string, string[]> = {
    background_and_need: ['problem_statement', 'market_analysis'],
    project_description: ['technology_description', 'company_description'],
    team_and_capacity: ['team_overview'],
    expected_results: ['impact_statement'],
    sustainability: ['sustainability_plan'],
    company_description: ['company_description'],
    executive_summary: ['company_description', 'problem_statement']
  };
  
  const relevantTypes = typeMapping[sectionType] || [];
  return blocks.filter(b => relevantTypes.includes(b.contentType));
}

export async function generateWithLibrary(
  grant: Grant,
  section: SectionTemplate,
  companyProfile: CompanyProfile,
  relevantBlocks: ContentBlock[]
): Promise<string> {
  const client = new Anthropic();
  
  const blocksContext = relevantBlocks.length > 0 
    ? `\nREUSABLE CONTENT FROM PREVIOUS APPLICATIONS (adapt these, don't copy verbatim):
${relevantBlocks.map(b => `[${b.contentType}]: ${b.content}`).join('\n\n')}`
    : '';
  
  const response = await client.messages.create({
    model: 'claude-opus-4-5',
    max_tokens: 1000,
    messages: [{
      role: 'user',
      content: `Write the "${section.title}" section for a grant application.

GRANT: ${grant.title} (${grant.source})
SECTION: ${section.description}
WORD LIMIT: ${section.maxWords} words
EVALUATION CRITERIA: ${section.evaluationCriteria}

COMPANY:
${companyProfile.name} | ${companyProfile.sector} | ${companyProfile.employees} employees
${companyProfile.description}
${blocksContext}

Write the section now. If reusable content was provided, adapt it to fit this 
specific grant's language and criteria — don't copy verbatim. Write in Swedish 
if this is a Swedish grant, English otherwise.
Output section text only, no headers.`
    }]
  });
  
  return response.content[0].type === 'text' ? response.content[0].text : '';
}
```
