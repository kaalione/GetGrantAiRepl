 03  Add sector mismatch penalty to scoring
Root cause
When the sector check returns 'no_match', the current code gives 0 points for that criterion. But 0 points still allows a law firm to accumulate enough points from geography, company type, and other passing criteria to exceed 60. Adding an explicit penalty for sector mismatches ensures that a fundamentally wrong industry match pulls the overall score down below the threshold.

// New (with penalty):
const SECTOR_MAX = 20;
const SECTOR_PENALTY = -15;
 
if (sectorCheck === 'match') {
 totalScore += SECTOR_MAX;
 breakdown.sector = { score: SECTOR_MAX, max: SECTOR_MAX,
   note: `Industry match: ${company.industry}` };
} else if (sectorCheck === 'partial') {
 totalScore += Math.round(SECTOR_MAX * 0.5);
 breakdown.sector = { score: 10, max: SECTOR_MAX,
   note: 'Partial sector match' };
} else if (sectorCheck === 'unknown') {
 totalScore += Math.round(SECTOR_MAX * 0.3);  // 30% for unknown
 breakdown.sector = { score: 6, max: SECTOR_MAX,
   note: 'Sector not specified in grant' };
} else {  // no_match — apply penalty
 totalScore += SECTOR_PENALTY;  // subtract 15 points
 breakdown.sector = { score: SECTOR_PENALTY, max: SECTOR_MAX,
   note: `Sector mismatch: grant needs ${eligibility.sectors.join(', ')}` };
 blockers.push(`Industry mismatch: your sector (${company.industry})`,
   `does not match grant focus (${eligibility.sectors.join(', ')})`);
}
