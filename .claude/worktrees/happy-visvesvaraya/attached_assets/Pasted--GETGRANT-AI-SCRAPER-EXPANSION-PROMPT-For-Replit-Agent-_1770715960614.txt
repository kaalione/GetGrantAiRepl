import requests
from bs4 import BeautifulSoup
from datetime import datetime
import time
import re

# Import from existing utils
from utils.db import upsert_grant, log_scrape_result

class [SourceName]Scraper:
    """
    Scraper for [Source Name]
    URL: [source URL]
    Method: BeautifulSoup / Playwright
    """
    
    def __init__(self, source_id=None):
        self.source_id = source_id
        self.base_url = "[BASE_URL]"
        self.source_name = "[Source Name]"
        self.headers = {
            'User-Agent': 'GetGrant.ai Bot/1.0 (grant aggregator; contact@getgrant.ai)',
            'Accept-Language': 'sv-SE,sv;q=0.9',
        }
    
    def fetch_grants_list(self):
        """Fetch the main grants listing page"""
        try:
            response = requests.get(
                self.base_url,
                headers=self.headers,
                timeout=30
            )
            response.raise_for_status()
            return BeautifulSoup(response.content, 'html.parser')
        except Exception as e:
            print(f"Error fetching {self.source_name}: {e}")
            return None
    
    def fetch_grant_detail(self, url):
        """Fetch individual grant detail page"""
        time.sleep(2)  # RATE LIMITING — minimum 2 seconds between requests
        try:
            response = requests.get(url, headers=self.headers, timeout=30)
            response.raise_for_status()
            return BeautifulSoup(response.content, 'html.parser')
        except Exception as e:
            print(f"Error fetching detail {url}: {e}")
            return None
    
    def parse_deadline(self, date_text):
        """Parse Swedish date formats to datetime"""
        if not date_text:
            return None
        
        date_text = date_text.strip()
        
        # Swedish month names
        swedish_months = {
            'januari': 1, 'februari': 2, 'mars': 3, 'april': 4,
            'maj': 5, 'juni': 6, 'juli': 7, 'augusti': 8,
            'september': 9, 'oktober': 10, 'november': 11, 'december': 12
        }
        
        # Try common formats
        # Format: "15 mars 2026"
        for month_name, month_num in swedish_months.items():
            if month_name in date_text.lower():
                try:
                    day = int(re.search(r'(\d{1,2})', date_text).group(1))
                    year_match = re.search(r'(20\d{2})', date_text)
                    year = int(year_match.group(1)) if year_match else datetime.now().year
                    return datetime(year, month_num, day)
                except:
                    pass
        
        # Format: "2026-03-15"
        try:
            return datetime.strptime(date_text[:10], '%Y-%m-%d')
        except:
            pass
        
        # Format: "15/3/2026" or "15.3.2026"
        for fmt in ['%d/%m/%Y', '%d.%m.%Y']:
            try:
                return datetime.strptime(date_text, fmt)
            except:
                pass
        
        return None
    
    def extract_amount(self, text):
        """Extract funding amounts from Swedish text"""
        if not text:
            return None, None, text
        
        amount_min = None
        amount_max = None
        
        # Look for patterns like "upp till 5 miljoner" or "max 500 000 kr"
        millions = re.findall(r'(\d+(?:[.,]\d+)?)\s*miljon(?:er)?', text.lower())
        if millions:
            for m in millions:
                val = float(m.replace(',', '.')) * 1_000_000
                if amount_max is None or val > amount_max:
                    amount_max = val
        
        # Look for "XXX 000 kr" or "XXX 000 SEK"
        thousands = re.findall(r'(\d[\d\s]*\d)\s*(?:kr|sek|kronor)', text.lower())
        if thousands:
            for t in thousands:
                val = int(t.replace(' ', ''))
                if amount_max is None or val > amount_max:
                    amount_max = val
        
        return amount_min, amount_max, text
    
    def determine_status(self, deadline, status_text=None):
        """Determine grant status based on deadline and text cues"""
        if status_text:
            status_lower = status_text.lower()
            if any(w in status_lower for w in ['stängd', 'avslutad', 'closed']):
                return 'closed'
            if any(w in status_lower for w in ['öppen', 'pågående', 'open', 'ansök nu']):
                return 'open'
            if any(w in status_lower for w in ['kommande', 'planerad', 'upcoming']):
                return 'upcoming'
        
        if deadline:
            if isinstance(deadline, str):
                deadline = self.parse_deadline(deadline)
            if deadline and deadline < datetime.now():
                return 'closed'
            return 'open'
        
        return 'open'  # Default to open if unknown
    
    def extract_keywords(self, title, description):
        """Extract relevant keywords from Swedish text"""
        keywords = []
        text = f"{title} {description}".lower()
        
        keyword_map = [
            'innovation', 'forskning', 'utveckling', 'hållbarhet',
            'export', 'internationalisering', 'digitalisering',
            'klimat', 'miljö', 'grön omställning', 'cirkulär ekonomi',
            'energi', 'förnybar', 'startup', 'entreprenörskap',
            'AI', 'artificiell intelligens', 'automation',
            'landsbygd', 'jordbruk', 'livsmedel', 'skog',
            'kultur', 'design', 'kreativ', 'media',
            'hälsa', 'vård', 'omsorg', 'välfärd',
            'transport', 'infrastruktur', 'bygg', 'bostad',

```python
"""
Deduplication utility for grant data.
Three levels:
1. URL match (exact) — handled by upsert_grant already
2. Title similarity (fuzzy) — catches variants
3. Content hash — catches identical content from different sources
"""

import hashlib
import re

def normalize_title(title):
    """Normalize title for comparison"""
    if not title:
        return ""
    title = title.lower().strip()
    title = re.sub(r'[^\w\s]', '', title)  # Remove punctuation
    title = re.sub(r'\s+', ' ', title)      # Normalize whitespace
    return title

def title_similarity(title1, title2):
    """Simple Levenshtein-like similarity ratio"""
    t1 = normalize_title(title1)
    t2 = normalize_title(title2)
    
    if not t1 or not t2:
        return 0.0
    
    if t1 == t2:
        return 1.0
    
    # Use set-based similarity (Jaccard)
    words1 = set(t1.split())
    words2 = set(t2.split())
    
    if not words1 or not words2:
        return 0.0
    
    intersection = words1 & words2
    union = words1 | words2
    
    return len(intersection) / len(union)

def content_hash(description):
    """Generate hash of normalized description"""
    if not description:
        return None
    normalized = re.sub(r'\s+', ' ', description.lower().strip())
    return hashlib.sha256(normalized.encode()).hexdigest()[:16]

def is_duplicate(new_grant, existing_grants, threshold=0.85):
    """
    Check if a grant is a duplicate of any existing grant.
    Returns (is_dup: bool, matching_grant_id: str or None)
    """
    new_title = normalize_title(new_grant.get('title', ''))
    new_hash = content_hash(new_grant.get('description', ''))
    
    for existing in existing_grants:
        # Level 1: URL match
        if new_grant.get('url') == existing.get('url'):
            return True, existing.get('id')
        
        # Level 2: Title similarity
        sim = title_similarity(new_grant.get('title', ''), existing.get('title', ''))
        if sim >= threshold:
            return True, existing.get('id')
        
        # Level 3: Content hash
        if new_hash and new_hash == content_hash(existing.get('description', '')):
            return True, existing.get('id')
    
    return False, None
```
