CRITICAL 1: AI Application Writing Module
Build an AI-powered grant application writing module for GetGrant.ai.

## Overview
Users select a grant, answer a few questions about their project, and the AI generates 
a complete draft application tailored to that specific grant's requirements and 
evaluation criteria.

## Backend: New API endpoints

Create `server/routes/applications.ts` with these endpoints:

### POST /api/applications/generate
Generate a draft application for a specific grant.

Request body:
{
  grantId: string,
  companyProfile: {
    name: string,
    description: string,
    sector: string,
    employees: number,
    revenue?: number,
    founded?: number,
    region: string
  },
  projectDescription: string,      // 200-1000 chars, what they want to do
  projectGoals: string,             // expected outcomes
  projectBudget: number,            // total project budget in SEK
  requestedAmount: number,          // amount applying for
  previousExperience?: string       // relevant prior work
}

Response:
{
  applicationId: string,
  grantId: string,
  grantTitle: string,
  sections: [
    {
      sectionKey: string,           // e.g. "project_description"
      sectionTitle: string,         // e.g. "Projektbeskrivning"
      content: string,              // AI-generated text
      wordCount: number,
      maxWords?: number,            // from grant template if known
      evaluationCriteria?: string   // what judges look for in this section
    }
  ],
  overallScore?: number,            // estimated fit score 0-100
  warnings: string[],               // e.g. "Budget exceeds maximum grant amount"
  generatedAt: string
}

### GET /api/applications/:id
Retrieve a saved application draft.

### PUT /api/applications/:id/section/:sectionKey
Update a specific section (user edits).

Request body: { content: string }

### POST /api/applications/:id/regenerate-section
Regenerate a specific section with new instructions.

Request body: { sectionKey: string, instructions: string }

### GET /api/applications/templates/:grantSource
Get the application template for a specific grant source.

```typescript
export async function generateApplication(
  grant: Grant,
  companyProfile: CompanyProfile,
  projectData: ProjectData
): Promise<GeneratedApplication> {
  const client = new Anthropic();
  
  // Determine which template to use based on grant source
  const sourceKey = grant.source.toLowerCase();
  const template = GRANT_TEMPLATES[sourceKey] || GRANT_TEMPLATES.generic;
  
  const sections = [];
  
  // Generate each section with a targeted prompt
  for (const sectionTemplate of template.sections) {
    const prompt = buildSectionPrompt(
      grant, 
      companyProfile, 
      projectData, 
      sectionTemplate
    );
    
    const response = await client.messages.create({
      model: 'claude-opus-4-5',
      max_tokens: 1500,
      messages: [{ role: 'user', content: prompt }]
    });
    
    const content = response.content[0].type === 'text' 
      ? response.content[0].text 
      : '';
    
    sections.push({
      sectionKey: sectionTemplate.key,
      sectionTitle: sectionTemplate.title,
      content: content.trim(),
      wordCount: content.split(' ').length,
      maxWords: sectionTemplate.maxWords,
      evaluationCriteria: sectionTemplate.evaluationCriteria
    });
  }
  
  // Generate overall warnings
  const warnings = generateWarnings(grant, companyProfile, projectData);
  
  return { sections, warnings };
}
```

🔴 CRITICAL 2: Eligibility Check
Build an AI-powered eligibility pre-check feature that automatically verifies 
whether a company meets a grant's requirements before they invest time applying.

## Overview
When a user views a grant or starts an application, they see an instant eligibility 
verdict: ✅ Likely eligible / ⚠️ Partially eligible / ❌ Likely ineligible — with 
clear explanations for each criterion.

## Backend: server/services/eligibilityChecker.ts
```typescript
import Anthropic from '@anthropic-ai/sdk';

interface EligibilityResult {
  verdict: 'eligible' | 'partial' | 'ineligible' | 'unknown';
  score: number;                    // 0-100 confidence
  criteria: EligibilityCriterion[];
  summary: string;                  // 1-2 sentence plain language verdict
  blockers: string[];               // hard blockers that disqualify
  warnings: string[];               // soft issues that may reduce chances
  strengths: string[];              // where the company is a strong fit
}

interface EligibilityCriterion {
  name: string;                     // e.g. "Company size"
  requirement: string;              // e.g. "Max 250 employees (SME)"
  companyValue: string;             // e.g. "45 employees"
  status: 'pass' | 'fail' | 'warning' | 'unknown';
  explanation: string;
}

export async function checkEligibility(
  grant: Grant,
  companyProfile: CompanyProfile
): Promise<EligibilityResult> {
  const client = new Anthropic();
  
  // First try structured eligibility data if available
  if (grant.eligibilityData) {
    const structuredResult = checkStructuredEligibility(
      grant.eligibilityData, 
      companyProfile
    );
    // If we got a confident result from structured data, enrich with AI summary
    if (structuredResult.score > 80 || structuredResult.blockers.length > 0) {
      structuredResult.summary = await generateEligibilitySummary(
        structuredResult, grant, companyProfile
      );
      return structuredResult;
    }
  }
  
  // Fall back to AI analysis of raw grant text
  const prompt = buildEligibilityPrompt(grant, companyProfile);
  
  const response = await client.messages.create({
    model: 'claude-opus-4-5',
    max_tokens: 1000,
    messages: [{ role: 'user', content: prompt }],
    system: `You are a Swedish grant eligibility expert. Analyze whether a company 
    meets a grant's requirements. Always respond with valid JSON only, no prose.`
  });
  
  const text = response.content[0].type === 'text' ? response.content[0].text : '{}';
  
  try {
    return JSON.parse(text) as EligibilityResult;
  } catch {
    return { verdict: 'unknown', score: 0, criteria: [], 
             summary: 'Could not determine eligibility', 
             blockers: [], warnings: [], strengths: [] };
  }
}

function checkStructuredEligibility(
  eligibilityData: GrantEligibilityData,
  company: CompanyProfile
): EligibilityResult {
  const criteria: EligibilityCriterion[] = [];
  const blockers: string[] = [];
  const warnings: string[] = [];
  const strengths: string[] = [];
  
  // Company size check
  if (eligibilityData.maxEmployees) {
    const pass = company.employees <= eligibilityData.maxEmployees;
    criteria.push({
      name: 'Company size',
      requirement: `Max ${eligibilityData.maxEmployees} employees`,
      companyValue: `${company.employees} employees`,
      status: pass ? 'pass' : 'fail',
      explanation: pass 
        ? `Your company (${company.employees} employees) is within the limit`
        : `Your company exceeds the maximum size requirement`
    });
    if (!pass) blockers.push(`Company too large (${company.employees} vs max ${eligibilityData.maxEmployees})`);
    else strengths.push('Company size qualifies as SME');
  }
  
  // Geography check
  if (eligibilityData.geographyRequirements?.length) {
    const geoMatch = eligibilityData.geographyRequirements.some(req => 
      req.toLowerCase().includes(company.region?.toLowerCase() || '') ||
      req.toLowerCase().includes('sweden') ||
      req.toLowerCase().includes('sverige')
    );
    criteria.push({
      name: 'Geographic eligibility',
      requirement: eligibilityData.geographyRequirements.join(', '),
      companyValue: company.region || 'Not specified',
      status: geoMatch ? 'pass' : 'warning',
      explanation: geoMatch 
        ? 'Your region appears to be covered'
        : 'Verify your region is covered by this grant'
    });
  }
  
  // Company type check
  if (eligibilityData.companyTypes?.length) {
    const typeMatch = eligibilityData.companyTypes.some(t => 
      t.toLowerCase().includes(company.companyType?.toLowerCase() || 'company')
    );
    criteria.push({
      name: 'Organization type',
      requirement: eligibilityData.companyTypes.join(' or '),
      companyValue: company.companyType || 'Company',
      status: typeMatch ? 'pass' : 'warning',
      explanation: typeMatch ? 'Your organization type is eligible' : 'Verify your organization type is covered'
    });
  }
  
  // Sector check
  if (eligibilityData.sectors?.length) {
    const sectorMatch = eligibilityData.sectors.some(s =>
      s.toLowerCase().includes(company.sector?.toLowerCase() || '')
    );
    if (sectorMatch) strengths.push(`Your sector (${company.sector}) is explicitly targeted`);
  }
  
  const failCount = criteria.filter(c => c.status === 'fail').length;
  const passCount = criteria.filter(c => c.status === 'pass').length;
  
  let verdict: EligibilityResult['verdict'];
  let score: number;
  
  if (failCount > 0) {
    verdict = 'ineligible';
    score = Math.max(0, 30 - (failCount * 15));
  } else if (criteria.length === 0) {
    verdict = 'unknown';
    score = 50;
  } else if (passCount === criteria.length) {
    verdict = 'eligible';
    score = 85 + Math.min(15, strengths.length * 5);
  } else {
    verdict = 'partial';
    score = 60;
  }
  
  return { verdict, score, criteria, summary: '', blockers, warnings, strengths };
}

function buildEligibilityPrompt(grant: Grant, company: CompanyProfile): string {
  return `Analyze whether this company is eligible for this grant.

GRANT:
Title: ${grant.title}
Source: ${grant.source}
Description: ${grant.description}
Eligibility text: ${grant.eligibilityText || 'Not available — infer from description'}
Max amount: ${grant.maxAmount || 'Not specified'}

COMPANY:
Name: ${company.name}
Sector: ${company.sector}
Employees: ${company.employees}
Annual revenue: ${company.revenue || 'Not provided'}
Region: ${company.region}
Company type: ${company.companyType || 'Private company (AB)'}
Founded: ${company.founded || 'Not provided'}
Description: ${company.description}

Respond with JSON only in this exact structure:
{
  "verdict": "eligible" | "partial" | "ineligible" | "unknown",
  "score": 0-100,
  "criteria": [
    {
      "name": "criterion name",
      "requirement": "what the grant requires",
      "companyValue": "company's actual value",
      "status": "pass" | "fail" | "warning" | "unknown",
      "explanation": "1 sentence explanation"
    }
  ],
  "summary": "1-2 sentence plain Swedish/English verdict",
  "blockers": ["hard disqualifier 1"],
  "warnings": ["soft concern 1"],
  "strengths": ["strong match point 1"]
}`;
}
```

🟠 HIGH 1: "Why does this match?" AI Explanation
Add transparent AI match explanations to every grant match result, showing users 
exactly why a grant was recommended for them.

## Context
Currently the AI matching returns a score but no explanation. Users don't know 
why a grant matched, which reduces trust and engagement. This adds a "Why this 
match?" feature inspired by Instrumentl's AI Prospecting Assistant.

```typescript
export async function generateMatchExplanation(
  grant: Grant,
  companyProfile: CompanyProfile,
  matchScore: number
): Promise<MatchExplanation> {
  const client = new Anthropic();
  
  const response = await client.messages.create({
    model: 'claude-haiku-4-5',  // Use Haiku for cost efficiency — this runs per grant
    max_tokens: 300,
    messages: [{
      role: 'user',
      content: `Explain in 2-3 bullet points why this grant matches this company.
Be specific — reference actual details from both. Be direct and concrete.
Write in the same language as the grant (Swedish for Swedish grants, English for EU grants).

GRANT: ${grant.title}
FUNDER: ${grant.source}  
GRANT FOCUS: ${grant.description?.substring(0, 300)}
SECTORS TARGETED: ${grant.sectors?.join(', ') || 'Not specified'}

COMPANY: ${companyProfile.name}
SECTOR: ${companyProfile.sector}
DESCRIPTION: ${companyProfile.description?.substring(0, 200)}
REGION: ${companyProfile.region}

MATCH SCORE: ${matchScore}/100

Respond with JSON:
{
  "headline": "One sentence summary of the match (max 15 words)",
  "reasons": [
    "Specific reason 1 — reference actual grant criteria and company detail",
    "Specific reason 2",
    "Specific reason 3 (optional)"
  ],
  "bestFitAspect": "The single strongest alignment point in 8 words or less"
}`
    }]
  });
  
  const text = response.content[0].type === 'text' ? response.content[0].text : '{}';
  try {
    return JSON.parse(text);
  } catch {
    return {
      headline: `Strong match for ${companyProfile.sector} companies`,
      reasons: ['Grant focus aligns with your sector', 'Your region is covered'],
      bestFitAspect: 'Sector alignment'
    };
  }
}
```

🟠 HIGH 2: Deadline Calendar + Push Notifications
Build a deadline calendar that shows upcoming deadlines for grants the user has 
bookmarked or that match their profile, with email and browser push notifications.

## Overview
Users can see all their relevant grant deadlines in one calendar view and set 
reminders. Inspired by Instrumentl's grant calendar and GrantHub.

## Backend

### New notification preferences schema

Add to schema.ts:
```typescript
export const notificationPreferences = pgTable('notification_preferences', {
  id: uuid('id').primaryKey().defaultRandom(),
  userId: varchar('user_id').notNull().unique(),
  emailEnabled: boolean('email_enabled').default(true),
  pushEnabled: boolean('push_enabled').default(false),
  reminderDaysBefore: integer('reminder_days_before').array().default([30, 14, 7, 1]),
  // 30 days, 14 days, 1 week, 1 day before deadline
  digestFrequency: varchar('digest_frequency').default('weekly'), // daily | weekly | never
  pushSubscription: jsonb('push_subscription'), // Web Push subscription object
  createdAt: timestamp('created_at').defaultNow(),
  updatedAt: timestamp('updated_at').defaultNow(),
});
```

### New API endpoints

**GET /api/calendar/events**
Returns calendar events for the logged-in user.

Query params:
- `start`: ISO date string (default: today)
- `end`: ISO date string (default: today + 3 months)
- `sources`: 'bookmarks' | 'matches' | 'all' (default: 'all')

Response:
```json

  "events": [
```

**PUT /api/notifications/preferences**
Update notification preferences.

**POST /api/notifications/push-subscribe**

### Scheduled notification job

Create `server/jobs/deadlineNotifier.ts`:
```typescript
import { CronJob } from 'cron';

// Run daily at 08:00 Stockholm time
export const deadlineNotifierJob = new CronJob(
  '0 8 * * *',
  async () => {
    const users = await getUsersWithNotificationsEnabled();
    
    for (const user of users) {
      const prefs = user.notificationPreferences;
      const grants = await getGrantsForUser(user.id); // bookmarks + top matches
      
      for (const grant of grants) {
        if (!grant.deadline) continue;
        
        const daysUntil = getDaysUntil(grant.deadline);
        
        // Check if today matches any of user's reminder thresholds
        if (prefs.reminderDaysBefore.includes(daysUntil)) {
          if (prefs.emailEnabled) {
            await sendDeadlineEmail(user, grant, daysUntil);
          }
          if (prefs.pushEnabled && prefs.pushSubscription) {
            await sendPushNotification(user.pushSubscription, {
              title: `Deadline: ${grant.title}`,
              body: `${daysUntil} day${daysUntil !== 1 ? 's' : ''} left to apply`,
              url: `/grants/${grant.id}`,
              icon: '/favicon.ico'
            });
          }
        }
      }
      
      // Weekly digest
      if (prefs.digestFrequency === 'weekly' && isMonday()) {
        const upcomingDeadlines = grants
          .filter(g => g.deadline && getDaysUntil(g.deadline) <= 30)
          .sort((a, b) => new Date(a.deadline).getTime() - new Date(b.deadline).getTime());
        
        if (upcomingDeadlines.length > 0) {
          await sendWeeklyDigestEmail(user, upcomingDeadlines);
        }
      }
    }
  },
  null,
  true,
  'Europe/Stockholm'
);
```
