Add re-extraction queue for low-confidence grants
Currently, grants that were extracted with confidence < 0.3 sit silently — the admin has no way to see them or trigger a retry. This task adds two things: a way to query the low-confidence grants per source, and an endpoint to re-queue them for extraction with the improved prompt.
 
1
Add a re-extraction endpoint to the admin API
In server/routes.ts, add a new admin endpoint alongside the existing
/api/admin/eligibility/extract endpoint.
 
// POST /api/admin/eligibility/reextract-low-confidence
// Re-runs extraction on grants where confidence_score < threshold
// Body: { threshold?: number, sourceName?: string, maxGrants?: number }
router.post('/eligibility/reextract-low-confidence',
 isAuthenticated, isAdminUser,
 async (req, res) => {
   const threshold = req.body.threshold ?? 0.3;
   const sourceName = req.body.sourceName;  // optional filter
   const maxGrants = req.body.maxGrants ?? 100;
 
   const whereClause = sourceName
     ? sql`status IN ('open','upcoming')
           AND source_name = ${sourceName}
           AND structured_eligibility IS NOT NULL
           AND (structured_eligibility->>'confidence_score')::float < ${threshold}`
     : sql`status IN ('open','upcoming')
           AND structured_eligibility IS NOT NULL
           AND (structured_eligibility->>'confidence_score')::float < ${threshold}`;
 
   const grantsToRetry = await db
     .select({ id: grants.id, title: grants.title })
     .from(grants)
     .where(whereClause)
     .limit(maxGrants);
 
   res.json({ queued: grantsToRetry.length, message: 'Re-extraction started' });
 
   // Run in background — don't await
   processGrantsInBackground(grantsToRetry.map(g => g.id));
 }
);
 
// GET /api/admin/eligibility/coverage
// Returns per-source coverage stats for the admin UI
router.get('/eligibility/coverage',
 isAuthenticated, isAdminUser,
 async (req, res) => {
   const stats = await db.execute(sql`
     SELECT
       source_name,
       COUNT(*) AS total,
       COUNT(*) FILTER (WHERE structured_eligibility IS NOT NULL
         AND (structured_eligibility->>'confidence_score')::float >= 0.3
       ) AS high_conf,
       COUNT(*) FILTER (WHERE structured_eligibility IS NULL) AS not_extracted,
       ROUND(100.0 * COUNT(*) FILTER (
         WHERE structured_eligibility IS NOT NULL
         AND (structured_eligibility->>'confidence_score')::float >= 0.3
       ) / COUNT(*), 0) AS pct_good
     FROM grants
     WHERE status IN ('open', 'upcoming')
     GROUP BY source_name
     ORDER BY pct_good ASC, not_extracted DESC
   `);
   res.json(stats.rows);
 }
);
