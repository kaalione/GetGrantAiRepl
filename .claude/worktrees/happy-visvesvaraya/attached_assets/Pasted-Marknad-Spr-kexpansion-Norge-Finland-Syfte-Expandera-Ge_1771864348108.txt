// client/src/lib/format.ts
export function formatAmount(amount: number, market: string): string {
  if (market === 'fi') return new Intl.NumberFormat('fi-FI', { style: 'currency', currency: 'EUR' }).format(amount);
  if (market === 'no') return new Intl.NumberFormat('nb-NO', { style: 'currency', currency: 'NOK' }).format(amount);
  return new Intl.NumberFormat('sv-SE', { style: 'currency', currency: 'SEK' }).format(amount);
}
