```typescript
import { Document, Packer, Paragraph, TextRun, AlignmentType, HeadingLevel } from 'docx';
import PDFDocument from 'pdfkit';

interface ApplicationData {
  grantTitle: string;
  companyName: string;
  content: string;
  generatedAt: Date;
}

export async function exportToDocx(application: ApplicationData): Promise<Buffer> {
  // Split content into paragraphs
  const paragraphs = application.content.split('\n\n').filter(p => p.trim());
  
  const doc = new Document({
    sections: [{
      properties: {
        page: {
          margin: {
            top: 1440, // 1 inch = 1440 twips
            right: 1440,
            bottom: 1440,
            left: 1440,
          },
        },
      },
      children: [
        // Title
        new Paragraph({
          text: application.grantTitle,
          heading: HeadingLevel.TITLE,
          alignment: AlignmentType.CENTER,
          spacing: {
            after: 400,
          },
        }),

        // Company name
        new Paragraph({
          text: `Ansökan från ${application.companyName}`,
          alignment: AlignmentType.CENTER,
          spacing: {
            after: 400,
          },
        }),

        // Date
        new Paragraph({
          text: new Date(application.generatedAt).toLocaleDateString('sv-SE', {
            year: 'numeric',
            month: 'long',
            day: 'numeric',
          }),
          alignment: AlignmentType.CENTER,
          spacing: {
            after: 800,
          },
          style: {
            font: {
              color: '666666',
              size: 20,
            },
          },
        }),

        // Main content
        ...paragraphs.map(para => {
          // Check if paragraph is a heading (starts with #)
          const isHeading = para.trim().startsWith('#');
          const text = isHeading ? para.replace(/^#+\s*/, '') : para;

          return new Paragraph({
            text: text,
            heading: isHeading ? HeadingLevel.HEADING_1 : undefined,
            spacing: {
              before: isHeading ? 400 : 200,
              after: 200,
            },
          });
        }),

        // Footer
        new Paragraph({
          text: '',
          spacing: { before: 800 },
        }),
        new Paragraph({
          text: 'Genererad med BidragAI',
          alignment: AlignmentType.CENTER,
          style: {
            font: {
              color: '999999',
              size: 18,
            },
          },
        }),
      ],
    }],
  });

  return await Packer.toBuffer(doc);
}

export async function exportToPdf(application: ApplicationData): Promise<Buffer> {
  return new Promise((resolve, reject) => {
    const chunks: Buffer[] = [];
    const doc = new PDFDocument({
      size: 'A4',
      margins: {
        top: 72,
        bottom: 72,
        left: 72,
        right: 72,
      },
    });

    // Collect PDF data
    doc.on('data', chunk => chunks.push(chunk));
    doc.on('end', () => resolve(Buffer.concat(chunks)));
    doc.on('error', reject);

    // Title
    doc.fontSize(24)
       .font('Helvetica-Bold')
       .text(application.grantTitle, { align: 'center' });

    doc.moveDown();

    // Company
    doc.fontSize(14)
       .font('Helvetica')
       .text(`Ansökan från ${application.companyName}`, { align: 'center' });

    doc.moveDown();

    // Date
    doc.fontSize(10)
       .fillColor('#666666')
       .text(
         new Date(application.generatedAt).toLocaleDateString('sv-SE', {
           year: 'numeric',
           month: 'long',
           day: 'numeric',
         }),
         { align: 'center' }
       );

    doc.moveDown(2);
    doc.fillColor('#000000');

    // Content
    const paragraphs = application.content.split('\n\n').filter(p => p.trim());

    paragraphs.forEach(para => {
      const isHeading = para.trim().startsWith('#');
      const text = isHeading ? para.replace(/^#+\s*/, '') : para;

      if (isHeading) {
        doc.fontSize(16)
           .font('Helvetica-Bold')
           .text(text, { align: 'left' });
        doc.moveDown(0.5);
      } else {
        doc.fontSize(12)
           .font('Helvetica')
           .text(text, {
             align: 'justify',
             lineGap: 4,
           });
        doc.moveDown();
      }
    });

    // Footer
    doc.fontSize(10)
       .fillColor('#999999')
       .text('Genererad med BidragAI', {
         align: 'center',
       });

    doc.end();
  });
}
```

```typescript
// Export as DOCX
router.get('/applications/:id/export/docx', requireAuth, async (req, res) => {
  try {
    const application = await db.query.applications.findFirst({
      where: and(
        eq(applications.id, req.params.id),
```

```typescript
import Stripe from 'stripe';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, {
  apiVersion: '2023-10-16',
});

export const PRICE_IDS = {
  pro: process.env.STRIPE_PRO_PRICE_ID!,
  enterprise: process.env.STRIPE_ENTERPRISE_PRICE_ID!,
};

export async function createCheckoutSession(
  userId: string,
  userEmail: string,
  priceId: string,
  successUrl: string,
  cancelUrl: string
) {
  const session = await stripe.checkout.sessions.create({
    customer_email: userEmail,
    client_reference_id: userId,
    mode: 'subscription',
    payment_method_types: ['card'],
    line_items: [
      {
        price: priceId,
        quantity: 1,
      },
    ],
    success_url: successUrl,
    cancel_url: cancelUrl,
    metadata: {
      userId,
    },
  });

  return session;
}

export async function createCustomerPortalSession(
  customerId: string,
  returnUrl: string
) {
  const session = await stripe.billingPortal.sessions.create({
    customer: customerId,
    return_url: returnUrl,
  });

  return session;
}

export async function cancelSubscription(subscriptionId: string) {
  await stripe.subscriptions.cancel(subscriptionId);
}

export async function getSubscription(subscriptionId: string) {
  return await stripe.subscriptions.retrieve(subscriptionId);
}

export { stripe };
```

STEP 5.6: Create Checkout Routes

Create server/routes/billing.ts:
```typescript
import express from 'express';
import { requireAuth } from '../middleware/auth';
import { createCheckoutSession, createCustomerPortalSession, PRICE_IDS } from '../services/stripe';

const router = express.Router();

// Create checkout session
router.post('/billing/checkout', requireAuth, async (req, res) => {
  try {
    const { plan } = req.body; // 'pro' or 'enterprise'

    if (!['pro', 'enterprise'].includes(plan)) {
      return res.status(400).json({ error: 'Invalid plan' });
    }

    const priceId = PRICE_IDS[plan];

    const session = await createCheckoutSession(
      req.user.id,
      req.user.email,
      priceId,
      `${process.env.APP_URL}/dashboard?checkout=success`,
      `${process.env.APP_URL}/pricing?checkout=canceled`
    );

    res.json({ url: session.url });
  } catch (error) {
    console.error('Checkout error:', error);
    res.status(500).json({ error: 'Failed to create checkout session' });
  }
});

// Create customer portal session (for managing subscription)
router.post('/billing/portal', requireAuth, async (req, res) => {
  try {
    const profile = await db.query.profiles.findFirst({
      where: eq(profiles.userId, req.user.id),
    });

    if (!profile?.stripeCustomerId) {
      return res.status(400).json({ error: 'No active subscription' });
    }

    const session = await createCustomerPortalSession(
      profile.stripeCustomerId,
      `${process.env.APP_URL}/dashboard`
    );

    res.json({ url: session.url });
  } catch (error) {
    console.error('Portal error:', error);
    res.status(500).json({ error: 'Failed to create portal session' });
  }
});

// Webhook to handle Stripe events
router.post('/billing/webhook', express.raw({ type: 'application/json' }), async (req, res) => {
  const sig = req.headers['stripe-signature'] as string;

  try {
    const event = stripe.webhooks.constructEvent(
      req.body,
      sig,
      process.env.STRIPE_WEBHOOK_SECRET!
    );

    switch (event.type) {
      case 'checkout.session.completed': {
        const session = event.data.object as Stripe.Checkout.Session;
        const userId = session.metadata?.userId;

        if (!userId) break;

        // Update user profile with subscription
        await db.update(profiles)
          .set({
            plan: session.metadata.plan || 'pro',
            stripeCustomerId: session.customer as string,
            stripeSubscriptionId: session.subscription as string,
            subscriptionStatus: 'active',
          })
          .where(eq(profiles.userId, userId));

        break;
      }

      case 'customer.subscription.updated':
      case 'customer.subscription.deleted': {
        const subscription = event.data.object as Stripe.Subscription;

        await db.update(profiles)
          .set({
            subscriptionStatus: subscription.status,
            subscriptionEndsAt: new Date(subscription.current_period_end * 1000),
            ...(subscription.status === 'canceled' && { plan: 'free' }),
          })
          .where(eq(profiles.stripeSubscriptionId, subscription.id));

        break;
      }
    }

    res.json({ received: true });
  } catch (error) {
    console.error('Webhook error:', error);
    res.status(400).send(`Webhook Error: ${error.message}`);
  }
});

export default router;
```

```typescript
export function requirePlan(minPlan: 'pro' | 'enterprise') {
  return async (req: Request, res: Response, next: NextFunction) => {
    const profile = await db.query.profiles.findFirst({
      where: eq(profiles.userId, req.user.id),
    });

    const planHierarchy = {
      free: 0,
      pro: 1,
      enterprise: 2,
    };

    const userPlanLevel = planHierarchy[profile?.plan || 'free'];
    const requiredPlanLevel = planHierarchy[minPlan];

    if (userPlanLevel < requiredPlanLevel) {
      return res.status(402).json({
        error: 'Upgrade required',
        message: `Denna funktion kräver ${minPlan.toUpperCase()}-plan`,
        upgradeUrl: '/pricing',
        currentPlan: profile?.plan || 'free',
        requiredPlan: minPlan,
      });
    }

    next();
  };
}
```

```typescript
router.patch('/applications/:id/status', requireAuth, async (req, res) => {
  try {
    const { status, notes, approvedAmount } = req.body;

    const application = await db.query.applications.findFirst({
      where: and(
        eq(applications.id, req.params.id),
        eq(applications.companyId, req.user.companyId)
      ),
    });

    if (!application) {
      return res.status(404).json({ error: 'Application not found' });
    }

    const updates: any = { status };

    if (status === 'submitted') {
      updates.submittedAt = new Date();
    }

    if (status === 'approved' || status === 'rejected') {
      updates.reviewedAt = new Date();
    }

    if (approvedAmount) {
      updates.approvedAmount = approvedAmount;
    }

    if (notes) {
      updates.notes = notes;
    }

    const [updated] = await db.update(applications)
      .set(updates)
      .where(eq(applications.id, req.params.id))
      .returning();

    res.json(updated);
  } catch (error) {
    console.error('Status update error:', error);
    res.status(500).json({ error: 'Failed to update status' });
  }
});
```
