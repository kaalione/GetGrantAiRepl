SETUP:
	1.	Add environment variable: ANTHROPIC_API_KEY
	2.	Install: npm install @anthropic-ai/sdk
	3.	Create lib/claude.ts: import Anthropic from '@anthropic-ai/sdk';

const client = new Anthropic({
  apiKey: process.env.ANTHROPIC_API_KEY,
});

export async function generateApplication(
  grant: Grant,
  company: Company,
  previousApplications?: string[] // For learning
) {
  const system = `Du är expert på svenska företagsbidrag. 
  Din uppgift är att skriva en övertygande ansökan baserat på:
  - Företagets information
  - Bidragets krav
  - Tidigare godkända ansökningar (om tillgängliga)
  
  Skriv professionellt på svenska med tydlig struktur.`;
  
  const prompt = `
  FÖRETAGSINFORMATION:
  Namn: ${company.company_name}
  Bransch: ${company.industry}
  Antal anställda: ${company.employees}
  Beskrivning: ${company.description}
  
  BIDRAG ATT SÖKA:
  Namn: ${grant.title}
  Beskrivning: ${grant.description}
  Krav: ${JSON.stringify(grant.eligibility_criteria)}
  Krav på ansökan: ${JSON.stringify(grant.application_requirements)}
  
  ${previousApplications ? `EXEMPEL PÅ GODKÄNDA ANSÖKNINGAR:\n${previousApplications.join('\n\n')}` : ''}
  
  Skriv en komplett ansökan som:
  1. Tydligt visar hur företaget uppfyller kraven
  2. Är övertygande och välformulerad
  3. Följer strukturen som krävs
  4. Är på korrekt svenska utan AI-känsla
  `;
  
  const message = await client.messages.create({
    model: 'claude-sonnet-4-5-20250929',
    max_tokens: 4000,
    system: system,
    messages: [{ role: 'user', content: prompt }],
  });
  
  return message.content[0].text;
}

export async function POST(request: Request) {
  const { grantId, companyId } = await request.json();
  
  // Fetch grant and company from DB
  const grant = await getGrant(grantId);
  const company = await getCompany(companyId);
  
  // Optional: Fetch similar successful applications
  // const examples = await getSimilarApplications(grant.source_name);
  
  const applicationText = await generateApplication(grant, company);
  
  // Save to applications table as draft
  const application = await createApplication({
    company_id: companyId,
    grant_id: grantId,
    generated_content: applicationText,
    ai_model_used: 'claude-sonnet-4-5',
    status: 'draft',
  });
  
  return Response.json({ 
    application,
    token_usage: 'estimated based on length' 
  });
}
