Create server/middleware/whitelabel.ts. This runs on every request
and injects partner branding context.
```typescript
import { Request, Response, NextFunction } from 'express';

// Cache partner lookups for 5 minutes to avoid DB hit on every request
const partnerCache = new Map<string, { partner: Partner | null; cachedAt: number }>();
const CACHE_TTL_MS = 5 * 60 * 1000;

export interface PartnerContext {
  isWhitelabel: boolean;
  partner: Partner | null;
  branding: BrandingConfig;
}

export interface BrandingConfig {
  platformName: string;
  logoUrl: string | null;
  faviconUrl: string | null;
  primaryColor: string;
  accentColor: string;
  primaryTextColor: string;
  fontFamily: string;
  tagline: string | null;
  supportEmail: string;
  supportUrl: string | null;
  showPoweredBy: boolean;
  footerText: string | null;
}

const DEFAULT_BRANDING: BrandingConfig = {
  platformName: 'GetGrant.ai',
  logoUrl: null,
  faviconUrl: null,
  primaryColor: '#2563EB',
  accentColor: '#10B981',
  primaryTextColor: '#FFFFFF',
  fontFamily: 'Inter',
  tagline: null,
  supportEmail: 'support@getgrant.ai',
  supportUrl: null,
  showPoweredBy: false,
  footerText: null,
};

export async function whitelabelMiddleware(
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> {
  const host = req.hostname;                         // e.g. "acme.getgrant.ai"
  
  // Parse subdomain
  const getgrantDomain = process.env.BASE_DOMAIN || 'getgrant.ai';
  const isSubdomain = host.endsWith(`.${getgrantDomain}`) && 
                       host !== getgrantDomain &&
                       host !== `www.${getgrantDomain}`;
  
  let lookupKey: string | null = null;
  let lookupType: 'subdomain' | 'customDomain' | null = null;
  
  if (isSubdomain) {
    lookupKey = host.replace(`.${getgrantDomain}`, '');
    lookupType = 'subdomain';
  } else if (host !== getgrantDomain && host !== `www.${getgrantDomain}`) {
    // Could be a custom domain
    lookupKey = host;
    lookupType = 'customDomain';
  }
  
  if (!lookupKey || !lookupType) {
    // Main domain — no white-label
    req.partnerContext = {
      isWhitelabel: false,
      partner: null,
      branding: DEFAULT_BRANDING
    };
    return next();
  }
  
  // Check cache
  const cacheKey = `${lookupType}:${lookupKey}`;
  const cached = partnerCache.get(cacheKey);
  if (cached && Date.now() - cached.cachedAt < CACHE_TTL_MS) {
    req.partnerContext = buildContext(cached.partner);
    return next();
  }
  
  // DB lookup
  try {
    const partner = await db.query.partners.findFirst({
      where: lookupType === 'subdomain'
        ? and(
            eq(partners.subdomain, lookupKey),
            eq(partners.status, 'active')
          )
        : and(
            eq(partners.customDomain, lookupKey),
            eq(partners.customDomainVerified, true),
            eq(partners.status, 'active')
          )
    });
    
    partnerCache.set(cacheKey, { partner: partner || null, cachedAt: Date.now() });
    req.partnerContext = buildContext(partner || null);
    next();
  } catch (error) {
    console.error('Whitelabel middleware error:', error);
    req.partnerContext = { isWhitelabel: false, partner: null, branding: DEFAULT_BRANDING };
    next();
  }
}

function buildContext(partner: Partner | null): PartnerContext {
  if (!partner) {
    return { isWhitelabel: false, partner: null, branding: DEFAULT_BRANDING };
  }
  
  return {
    isWhitelabel: true,
    partner,
    branding: {
      platformName: partner.platformName || partner.companyName,
      logoUrl: partner.logoUrl,
      faviconUrl: partner.faviconUrl,
      primaryColor: partner.primaryColor || DEFAULT_BRANDING.primaryColor,
      accentColor: partner.accentColor || DEFAULT_BRANDING.accentColor,
      primaryTextColor: partner.primaryTextColor || DEFAULT_BRANDING.primaryTextColor,
      fontFamily: partner.fontFamily || DEFAULT_BRANDING.fontFamily,
      tagline: partner.tagline,
      supportEmail: partner.supportEmail || partner.contactEmail,
      supportUrl: partner.supportUrl,
      showPoweredBy: partner.showPoweredBy,
      footerText: partner.footerText,
    }
  };
}

// Extend Express Request type
declare global {
  namespace Express {
    interface Request {
      partnerContext: PartnerContext;
    }
  }
}

// Call invalidatePartnerCache(partnerId) when partner settings are updated
export function invalidatePartnerCache(partnerId: string): void {
  for (const [key, value] of partnerCache.entries()) {
    if (value.partner?.id === partnerId) {
      partnerCache.delete(key);
    }
  }
}
```

## 3.1 — Partner registration & setup
POST /api/partners/register
Register as a new partner.
Body: {
companyName: string,
companyOrgNumber?: string,
companyWebsite?: string,
contactName: string,
contactEmail: string,
contactPhone?: string,
subdomain: string,           // validated: 3-30 chars, [a-z0-9-], no reserved words

Reserved: ['www', 'app', 'api', 'admin', 'mail', 'smtp', 'ftp',

Create partner record with status: 'pending_setup'

Send admin notification to GetGrant.ai admin

Response: {

Body: { companyName?, contactName?, contactEmail?,

Process:

Create GetGrant.ai user account
Create partnerClients record with status: 'active'
Apply clientPlan entitlement
If inviteToken: mark original invite as accepted
Send welcome email (branded with partner branding)
Log activity: 'client_joined'

Store bcrypt hash of full key

Return FULL KEY once — never shown again

```typescript
const RESERVED_SUBDOMAINS = new Set([
  'www', 'app', 'api', 'admin', 'mail', 'smtp', 'ftp',
  'getgrant', 'support', 'help', 'blog', 'dev', 'staging',
  'partner', 'partners', 'static', 'cdn', 'assets',

function validateSubdomain(subdomain: string): { valid: boolean; reason?: string } {
  if (subdomain.length < 3 || subdomain.length > 30)
    return { valid: false, reason: 'Subdomänen måste vara 3–30 tecken' };
  if (!/^[a-z0-9][a-z0-9-]*[a-z0-9]$/.test(subdomain))
    return { valid: false, reason: 'Endast gemener, siffror och bindestreck' };
  if (RESERVED_SUBDOMAINS.has(subdomain))
    return { valid: false, reason: 'Denna subdomain är reserverad' };
  return { valid: true };
}

function validateCustomDomain(domain: string): boolean {
  // Must be a valid hostname, not a getgrant.ai domain

         !domain.endsWith('getgrant.ai') &&
```
