Quick implementation:
sql-- Add to profiles table
ALTER TABLE profiles ADD COLUMN plan TEXT DEFAULT 'free';
-- 'free', 'pro', 'enterprise'
typescript// Middleware
function requirePlan(minPlan: 'pro' | 'enterprise') {
  return (req, res, next) => {
    const userPlan = req.user.profile.plan;
    if (!hasAccess(userPlan, minPlan)) {
      return res.status(402).json({
        error: 'Upgrade required',
        message: 'Denna funktion kräver Pro-plan',
        upgradeUrl: '/pricing'
      });
    }
    next();
  };
}

// Apply to routes
app.post('/api/notifications/enable', requireAuth, requirePlan('pro'), ...);

Break even vid ~15 pro users (täcker AI costs)


PRIO 3: Application Export Formats (2h)
Problem: "Download" är nämnt men inte vilket format.
Bidragsansökningar måste oftast:

Vara i Word (.docx) för att myndigheter ska kunna kommentera
Eller PDF för submission
Ha proper formatting (headers, sections, page breaks)

Förslag:
A) Export till DOCX:
bashnpm install docx
typescriptimport { Document, Packer, Paragraph, TextRun } from 'docx';

async function exportToDocx(application) {
  const doc = new Document({
    sections: [{
      properties: {},
      children: [
        new Paragraph({
          children: [
            new TextRun({
              text: application.grant.title,
              bold: true,
              size: 32,
            }),
          ],
        }),
        new Paragraph({ text: '' }), // Line break
        new Paragraph({
          children: [
            new TextRun({
              text: 'ANSÖKAN',
              bold: true,
              size: 28,
            }),
          ],
        }),
        new Paragraph({ text: '' }),
        ...application.generatedContent.split('\n').map(
          para => new Paragraph({ text: para })
        ),
      ],
    }],
  });

  const buffer = await Packer.toBuffer(doc);
  return buffer;
}

// Route
app.get('/api/applications/:id/export/docx', requireAuth, async (req, res) => {
  const app = await getApplication(req.params.id, req.user.id);
  const buffer = await exportToDocx(app);
  
  res.set({
    'Content-Type': 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    'Content-Disposition': `attachment; filename="ansokan-${app.grant.title}.docx"`,
  });
  res.send(buffer);
});
B) Export till PDF:
bashnpm install pdfkit
Liknande implementation.
Värde: Detta är nästan ett krav - ingen kommer skicka in en ansökan som plain text.
