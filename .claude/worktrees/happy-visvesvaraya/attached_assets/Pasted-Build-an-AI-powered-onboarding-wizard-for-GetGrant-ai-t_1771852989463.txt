Build an AI-powered onboarding wizard for GetGrant.ai that lets users enter
their company website URL and automatically populates their company profile
using AI. This replaces the current manual form-filling experience with an
intelligent, conversational flow that dramatically reduces time-to-value.

// ─── URL normalization ────────────────────────────────────────────────────

export function normalizeUrl(input: string): string {
  let url = input.trim();
  
  // Add protocol if missing
  if (!url.startsWith('http://') && !url.startsWith('https://')) {
    url = `https://${url}`;
  }
  
  try {
    const parsed = new URL(url);
    // Normalize: remove trailing slash, force https
    return `https://${parsed.hostname}${parsed.pathname.replace(/\/$/, '')}`;
  } catch {
    throw new Error(`Ogiltig URL: ${input}`);
  }
}

// ─── Multi-page scraping ──────────────────────────────────────────────────

export async function scrapeWebsite(
  baseUrl: string
): Promise<{ content: string; pagesScraped: string[]; errors: string[] }> {
  const normalizedBase = normalizeUrl(baseUrl);
  const hostname = new URL(normalizedBase).hostname;
  
  // Pages to try scraping (in order of information richness)
  const pagesToTry = [
    normalizedBase,                          // homepage
    `${normalizedBase}/om-oss`,              // Swedish "about us"
    `${normalizedBase}/about`,
    `${normalizedBase}/about-us`,
    `${normalizedBase}/om`,
    `${normalizedBase}/foretaget`,
    `${normalizedBase}/vi`,
    `${normalizedBase}/team`,
  ].filter((url, index, self) => self.indexOf(url) === index); // deduplicate

  const scrapedContent: string[] = [];
  const pagesScraped: string[] = [];
  const errors: string[] = [];
  
  const SCRAPE_TIMEOUT_MS = 8000;
  const MAX_CONTENT_LENGTH = 15000;          // chars per page
  
  for (const pageUrl of pagesToTry) {
    // Stop if we have enough content
    if (scrapedContent.join('\n').length > 30000) break;
    
    // Check cache first
    const cached = await db.query.websiteScrapeCache.findFirst({
      where: and(
        eq(websiteScrapeCache.normalizedUrl, pageUrl),
        gt(websiteScrapeCache.expiresAt, new Date())
      )
    });
    
    if (cached?.scrapedContent) {
      scrapedContent.push(`[${pageUrl}]\n${cached.scrapedContent}`);
      pagesScraped.push(pageUrl);
      continue;
    }
    
    try {
      const controller = new AbortController();
      const timeout = setTimeout(() => controller.abort(), SCRAPE_TIMEOUT_MS);
      
      const response = await fetch(pageUrl, {
        signal: controller.signal,
        headers: {
          'User-Agent': 'Mozilla/5.0 (compatible; GetGrant.ai/1.0; +https://getgrant.ai/bot)',
          'Accept': 'text/html,application/xhtml+xml',
          'Accept-Language': 'sv-SE,sv;q=0.9,en;q=0.8'
        }
      });
      clearTimeout(timeout);
      
      if (!response.ok) {
        // 404 on /om-oss is expected and not an error worth logging
        if (response.status !== 404) {
          errors.push(`${pageUrl}: HTTP ${response.status}`);
        }
        continue;
      }
      
      const html = await response.text();
      const extracted = extractTextFromHtml(html, MAX_CONTENT_LENGTH);
      
      // Cache the result
      const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000);
      await db.insert(websiteScrapeCache).values({
        url: pageUrl,
        normalizedUrl: pageUrl,
        scrapedContent: extracted,
        statusCode: response.status,
        contentLength: extracted.length,
        expiresAt
      }).onConflictDoUpdate({
        target: websiteScrapeCache.url,
        set: { scrapedContent: extracted, scrapedAt: new Date(), expiresAt }
      });
      
      scrapedContent.push(`[${pageUrl}]\n${extracted}`);
      pagesScraped.push(pageUrl);
      
    } catch (error: any) {
      if (error.name === 'AbortError') {
        errors.push(`${pageUrl}: Timeout efter ${SCRAPE_TIMEOUT_MS / 1000}s`);
      } else if (pageUrl !== normalizedBase) {
        // Silently skip sub-pages that fail
      } else {
        errors.push(`${pageUrl}: ${error.message}`);
      }
    }
    
    // Rate limit: wait 500ms between requests to same domain
    await new Promise(r => setTimeout(r, 500));
  }
  
  return {
    content: scrapedContent.join('\n\n---\n\n'),
    pagesScraped,
    errors
  };
}

function extractTextFromHtml(html: string, maxLength: number): string {
  const $ = cheerio.load(html);
  
  // Remove non-content elements
  $('script, style, nav, footer, header, aside, iframe, noscript, ' +
    '.cookie-banner, .cookie-notice, #cookie, [aria-hidden="true"]').remove();
  
  // Prioritize semantic content areas
  const contentSelectors = [
    'main', 'article', '[role="main"]',
    '.about, .om-oss, .company, .foretag',
    'h1, h2, h3', 'p', 'li'
  ];
  
  let text = '';
  
  // Extract meta description first (often best summary)
  const metaDescription = $('meta[name="description"]').attr('content');
  if (metaDescription) {
    text += `[Meta description]: ${metaDescription}\n\n`;
  }
  
  // Extract title
  const title = $('title').text().trim();
  if (title) text += `[Title]: ${title}\n\n`;
  
  // Extract body text
  text += $('body').text()
    .replace(/\s+/g, ' ')
    .replace(/\n{3,}/g, '\n\n')
    .trim();
  
  return text.substring(0, maxLength);
}

Start or resume an onboarding session for the current user.
Process:

Check if existing incomplete session → return it

If user already has a company profile, set currentStep to the

currentStep: number,
completedAt: string | null,

}
POST /api/onboarding/extract
Main endpoint — scrape website and extract company data.
Body: { sessionId: string, websiteUrl: string }
Validation:

Session must belong to the current user
URL must be valid and not a localhost/private IP
Rate limit: max 3 extraction attempts per session, 10 per user per day

Process:

Update session: extractionStatus='scraping', websiteUrl=normalizedUrl
Scrape website (scrapeWebsite())
Update session: extractionStatus='extracting'
Extract company data with AI (extractCompanyDataFromWebsite())
Store raw result in session.rawExtractedData
Store confidence scores in session.confidenceScores
Update session: extractionStatus='success' (or 'partial' if many nulls)
Increment extractionAttempts

Response: {
sessionId: string,
status: 'success' | 'partial' | 'failed',
extractedData: ExtractedCompanyData,
mappedProfile: Partial<CompanyProfile>,
fieldsFound: number,           // how many non-null fields extracted
totalFields: number,
confidenceScores: Record<string, number>,
pagesScraped: string[],
message: string                // human-readable summary
}
Error responses:

400: Invalid URL
400: URL appears to be localhost or private IP
429: Too many extraction attempts
422: Website unreachable or returned no content
{ error: 'website_unreachable', message: string, suggestion: string }
504: Scraping timed out
{ error: 'timeout', message: string }

GET /api/onboarding/session
Get current session state for the logged-in user.
Response: full session object including rawExtractedData
PUT /api/onboarding/step
Update the current step.

POST /api/onboarding/save-profile
Save the reviewed/edited profile data to the companies table.
Body: {
sessionId: string,
profileData: CompanyProfile,

}
Process:

Validate all required fields present
Upsert company profile in companies table
Update session: currentStep = 5 (sector/goals step)
Track which fields were auto-filled vs edited

Response: { companyId: string, profileCompletion: number }
POST /api/onboarding/complete
Mark onboarding as complete.
Body: { sessionId: string, extractionRating?: number, extractionFeedback?: string }
Process:

Set session.completedAt = now()
Store user feedback if provided

Sets session.skippedAt = now()
POST /api/onboarding/retry-extraction
Retry extraction with the same or a different URL.
Body: { sessionId: string, websiteUrl?: string }

// In the post-login redirect handler:

async function getPostLoginRedirect(userId: string): Promise<string> {
  const company = await getUserCompany(userId);
  
  if (!company) {
    // Brand new user — send to onboarding
    return '/onboarding';
  }
  
  const profileCompletion = calculateProfileCompletion(company);
  
  if (profileCompletion < 40) {
    // Very incomplete profile — nudge to onboarding
    const session = await getOrCreateOnboardingSession(userId, company.id);
    return `/onboarding?resume=${session.id}`;
  }
  
  // Profile exists and is reasonable — go to dashboard
  return '/dashboard';
}
