```typescript
interface CompanyProfile {
  companyName?: string;
  orgNumber?: string;
  industry?: string;
  employees?: number;
  revenue?: number;
  foundedYear?: number;
  location?: string;
  description?: string;
  focusAreas?: string[];
  contactEmail?: string;
}

interface ProfileCompletionResult {
  percentage: number;
  missingFields: string[];
  completedFields: string[];
  tier: 'incomplete' | 'basic' | 'good' | 'excellent';
}

export function calculateProfileCompletion(profile: CompanyProfile): ProfileCompletionResult {
  const fields = {
    companyName: { weight: 15, label: 'Företagsnamn' },
    orgNumber: { weight: 5, label: 'Organisationsnummer' },
    industry: { weight: 15, label: 'Bransch' },
    employees: { weight: 10, label: 'Antal anställda' },
    revenue: { weight: 10, label: 'Omsättning' },
    foundedYear: { weight: 5, label: 'Grundat år' },
    location: { weight: 10, label: 'Plats' },
    description: { weight: 20, label: 'Beskrivning' },
    focusAreas: { weight: 10, label: 'Fokusområden' },
    contactEmail: { weight: 5, label: 'Kontaktemail' },
  };

  let totalScore = 0;
  const missingFields: string[] = [];
  const completedFields: string[] = [];

  Object.entries(fields).forEach(([key, config]) => {
    const value = profile[key as keyof CompanyProfile];
    const hasValue = Array.isArray(value) 
      ? value.length > 0 
      : value !== null && value !== undefined && value !== '';

    if (hasValue) {
      totalScore += config.weight;
      completedFields.push(config.label);
    } else {
      missingFields.push(config.label);
    }
  });

  let tier: ProfileCompletionResult['tier'] = 'incomplete';
  if (totalScore >= 90) tier = 'excellent';
  else if (totalScore >= 70) tier = 'good';
  else if (totalScore >= 50) tier = 'basic';

  return {
    percentage: totalScore,
    missingFields,
    completedFields,
    tier,
  };
}

export function getProfileCompletionBenefits(percentage: number) {
  if (percentage >= 90) {
    return {
      matchQuality: 'excellent',
      expectedMatches: '15-20',
      aiAccuracy: '95%+',
      message: 'Perfekt! Ni får de bästa möjliga matchningarna.',
    };
  } else if (percentage >= 70) {
    return {
      matchQuality: 'good',
      expectedMatches: '10-15',
      aiAccuracy: '85-90%',
      message: 'Bra! Lägg till några fält till för ännu bättre matchningar.',
    };
  } else if (percentage >= 50) {
    return {
      matchQuality: 'basic',
      expectedMatches: '5-10',
      aiAccuracy: '70-80%',
      message: 'OK matchningar, men komplettera profilen för bättre resultat.',
    };
  } else {
    return {
      matchQuality: 'poor',
      expectedMatches: '0-5',
      aiAccuracy: '50-60%',
      message: 'Begränsade matchningar. Komplettera profilen för att låsa upp full potential.',
    };
  }
}
```

STEP 2.2: Create Profile Completion Alert Component

Create: `src/components/ProfileCompletionAlert.tsx`
```typescript
import { useState, useEffect } from 'react';
import { Alert, AlertDescription, AlertTitle } from '@/components/ui/alert';
import { Button } from '@/components/ui/button';
import { Progress } from '@/components/ui/progress';
import { AlertCircle, TrendingUp, CheckCircle, Sparkles } from 'lucide-react';
import { calculateProfileCompletion, getProfileCompletionBenefits } from '@/lib/profileCompletion';

interface ProfileCompletionAlertProps {
  profile: any;
  onComplete?: () => void;
}

export function ProfileCompletionAlert({ profile, onComplete }: ProfileCompletionAlertProps) {
  const [completion, setCompletion] = useState({ percentage: 0, missingFields: [], tier: 'incomplete' as const });
  const [benefits, setBenefits] = useState({ matchQuality: 'poor', expectedMatches: '0', aiAccuracy: '0%', message: '' });

  useEffect(() => {
    if (profile) {
      const result = calculateProfileCompletion(profile);
      const benefitsInfo = getProfileCompletionBenefits(result.percentage);
      setCompletion(result);
      setBenefits(benefitsInfo);
    }
  }, [profile]);

  // Don't show if profile is excellent
  if (completion.percentage >= 90) return null;

  const getAlertVariant = () => {
    if (completion.percentage < 50) return 'destructive';
    if (completion.percentage < 70) return 'warning';
    return 'default';
  };

  const getIcon = () => {
    if (completion.percentage < 50) return AlertCircle;
    if (completion.percentage < 70) return TrendingUp;
    return Sparkles;
  };

  const Icon = getIcon();

  return (
    <Alert className={`
      ${completion.percentage < 50 ? 'border-red-500 bg-red-50 dark:bg-red-900/20' : ''}
      ${completion.percentage >= 50 && completion.percentage < 70 ? 'border-amber-500 bg-amber-50 dark:bg-amber-900/20' : ''}
      ${completion.percentage >= 70 ? 'border-blue-500 bg-blue-50 dark:bg-blue-900/20' : ''}
    `}>
      <Icon className="h-5 w-5" />
      <AlertTitle className="flex items-center justify-between mb-3">
        <span>Komplettera din profil för bättre matchningar</span>
        <span className="text-2xl font-bold">{completion.percentage}%</span>
      </AlertTitle>
      <AlertDescription className="space-y-4">
        <Progress value={completion.percentage} className="h-2" />
        
        <div className="text-sm space-y-2">
          <p className="font-medium">{benefits.message}</p>
          
          <div className="grid grid-cols-3 gap-4 p-3 rounded-lg bg-white/50 dark:bg-gray-800/50">
            <div>
              <div className="text-xs text-gray-600 dark:text-gray-400">Förväntade matchningar</div>
              <div className="text-lg font-bold text-blue-600">{benefits.expectedMatches}</div>
            </div>
            <div>
              <div className="text-xs text-gray-600 dark:text-gray-400">AI-noggrannhet</div>
              <div className="text-lg font-bold text-green-600">{benefits.aiAccuracy}</div>
            </div>
            <div>
              <div className="text-xs text-gray-600 dark:text-gray-400">Matchningskvalitet</div>
              <div className="text-lg font-bold capitalize">{benefits.matchQuality}</div>
            </div>
          </div>

          {completion.missingFields.length > 0 && (
            <div>
              <p className="font-medium mb-1">Fält som saknas:</p>
              <div className="flex flex-wrap gap-2">
                {completion.missingFields.slice(0, 5).map(field => (
                  <span
                    key={field}
                    className="text-xs px-2 py-1 rounded-full bg-gray-200 dark:bg-gray-700"
                  >
                    {field}
                  </span>
                ))}
                {completion.missingFields.length > 5 && (
                  <span className="text-xs px-2 py-1 rounded-full bg-gray-200 dark:bg-gray-700">
                    +{completion.missingFields.length - 5} fler
                  </span>
                )}
              </div>
            </div>
          )}
        </div>

        <div className="flex gap-2">
          <Button 
            size="sm" 
            onClick={() => window.location.href = '/profile'}
            className="bg-blue-600 hover:bg-blue-700"
          >
            Komplettera profil (tar 2 min)
          </Button>
          {onComplete && (
            <Button size="sm" variant="ghost" onClick={onComplete}>
              Påminn mig senare
            </Button>
          )}
        </div>
      </AlertDescription>
    </Alert>
  );
}
```

// In server/routes/grants.ts

router.get('/grants/:id', requireAuth, async (req, res) => {
  try {
    const grant = await db.query.grants.findFirst({
      where: eq(grants.id, req.params.id),
    });

    if (!grant) {
      return res.status(404).json({ error: 'Grant not found' });
    }

    // Get user's company
    const company = await db.query.companies.findFirst({
      where: eq(companies.userId, req.user.id),
    });

    let matchAnalysis = null;

    if (company) {
      // Check if we have a cached analysis
      const existingAnalysis = await db.query.grantMatches.findFirst({
        where: and(
          eq(grantMatches.grantId, grant.id),
          eq(grantMatches.companyId, company.id)
        ),
      });

      if (existingAnalysis) {
        matchAnalysis = {
          score: existingAnalysis.matchScore,
          reasoning: existingAnalysis.reasoning,
          strengths: existingAnalysis.strengths || [],
          concerns: existingAnalysis.concerns || [],
        };
      } else {
        // Calculate basic rule-based match
        const basicMatch = calculateBasicMatch(company, grant);
        matchAnalysis = {
          score: basicMatch.score,
          reasoning: basicMatch.reasoning,
          strengths: basicMatch.strengths,
          concerns: basicMatch.concerns,
        };
      }
    }

    res.json({
      grant,
      matchAnalysis,
    });
  } catch (error) {
    console.error('Grant detail error:', error);
    res.status(500).json({ error: 'Failed to fetch grant' });
  }
});

function calculateBasicMatch(company, grant) {
  const strengths = [];
  const concerns = [];
  let score = 50; // Base score

  // Industry match
  if (grant.targetGroup?.includes(company.industry)) {
    score += 20;
    strengths.push(`Er bransch (${company.industry}) matchar bidragets målgrupp`);
  } else {
    concerns.push('Bidraget riktar sig till andra branscher');
  }

  // Company size match
  const employeeCount = company.employees || 0;
  if (grant.targetGroup?.includes('SME') || grant.targetGroup?.includes('små och medelstora företag')) {
    if (employeeCount < 250) {
      score += 15;
      strengths.push('Ni är ett SME vilket passar bidragskraven');
    } else {
      concerns.push('Bidraget riktar sig främst till mindre företag');
    }
  }

  // Keywords match
  const companyKeywords = (company.description?.toLowerCase() || '').split(' ');
  const grantKeywords = (grant.keywords || []).map(k => k.toLowerCase());
  const matchingKeywords = grantKeywords.filter(k => 
    companyKeywords.some(ck => ck.includes(k) || k.includes(ck))
  );

  if (matchingKeywords.length > 0) {
    score += Math.min(15, matchingKeywords.length * 5);
    strengths.push(`Nyckelord matchar: ${matchingKeywords.join(', ')}`);
  }

  // Location match
  if (grant.region && company.location) {
    if (grant.region.toLowerCase().includes(company.location.toLowerCase())) {
      score += 10;
      strengths.push('Ni är lokaliserade i bidragets region');
    }
  }

  const reasoning = strengths.length > 0
    ? `Detta bidrag passar er eftersom ${strengths.join(', och ')}.`
    : 'Detta bidrag kan passa men vi rekommenderar att köra en AI-analys för att få exakta detaljer.';

  return {
    score: Math.min(100, score),
    reasoning,
    strengths,
    concerns,
  };
}
