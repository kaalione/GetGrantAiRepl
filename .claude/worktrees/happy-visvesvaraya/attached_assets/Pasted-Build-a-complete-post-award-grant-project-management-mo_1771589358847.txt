Build a complete post-award grant project management module for GetGrant.ai.
This is a self-contained feature that lets users track everything after a grant
is approved: milestones, deliverables, reports, budget, and team tasks.

import Anthropic from '@anthropic-ai/sdk';

const FUNDER_REPORT_STYLES: Record<string, string> = {
  vinnova: `
    Vinnova progress reports should:
    - Be written in Swedish
    - Use formal but clear language
    - Structure: Sammanfattning → Genomförda aktiviteter → Resultat → 
      Avvikelser och åtgärder → Nästa period → Ekonomisk redovisning
    - Reference the original project goals and show measurable progress
    - Use TRL (Technology Readiness Level) language for technical projects
    - Quantify outputs wherever possible (number of tests, prototypes, users, etc.)
    - Keep to 2-4 pages for progress reports, 5-8 for interim/final
  `,
  tillvaxtverket: `
    Tillväxtverket reports should:
    - Be written in Swedish
    - Focus on regional impact and job creation
    - Structure: Projektöversikt → Aktiviteter → Effekter och resultat → 
      Ekonomisk redovisning → Fortsatt arbete
    - Include concrete numbers: jobs created/secured, companies reached,
      participants trained, etc.
    - Reference the regional development strategy alignment
    - Highlight sustainability aspects
  `,
  energimyndigheten: `
    Energimyndigheten reports should:
    - Be written in Swedish  
    - Emphasize energy/climate measurables: kWh saved, CO2 reduced, efficiency %
    - Structure: Teknisk rapport → Energiberäkningar → Marknadspotential → 
      Ekonomi → Nästa steg
    - Include TRL progression
    - Reference Swedish energy transition goals (Klimatmålen)
  `,
  generic: `
    Report should:
    - Match the language of the grant (Swedish for Swedish funders, English for EU)
    - Be professional and factual
    - Structure: Executive Summary → Activities Completed → Results → 
      Challenges & Mitigations → Next Steps → Financial Overview
    - Be specific — avoid vague statements, use numbers
  `
};

export async function generateProjectReport(
  project: GrantProject,
  report: ProjectReport,
  milestones: ProjectMilestone[],
  budgetCategories: ProjectBudgetCategory[],
  teamMembers: ProjectTeamMember[],
  originalApplication: Application | null,
  options: {
    includeFinancials: boolean;
    customInstructions?: string;
  }
): Promise<{ content: string; executiveSummary: string; financialSummary?: string }> {
  const client = new Anthropic();
  
  const completedMilestones = milestones.filter(m => m.status === 'completed');
  const pendingMilestones = milestones.filter(m => m.status === 'pending' || m.status === 'in_progress');
  const delayedMilestones = milestones.filter(m => m.status === 'delayed');
  
  // Budget summary
  const totalBudgeted = budgetCategories.reduce((sum, c) => sum + (c.budgetedAmountSek || 0), 0);
  const totalSpent = budgetCategories.reduce((sum, c) => sum + (c.spentAmountSek || 0), 0);
  const budgetText = budgetCategories
    .map(c => `${c.categoryLabel || c.category}: ${c.spentAmountSek?.toLocaleString('sv-SE')} / ${c.budgetedAmountSek?.toLocaleString('sv-SE')} SEK (${Math.round((c.spentAmountSek || 0) / (c.budgetedAmountSek || 1) * 100)}%)`)

  });

  try {
    const result = JSON.parse(text.replace(/```json|```/g, '').trim());
    return {
      content: result.content || '',
      executiveSummary: result.executiveSummary || '',
      financialSummary: result.financialSummary || undefined
    };
  } catch {
    // If JSON parse fails, return the raw text as content
    return {
      content: text,
      executiveSummary: text.split('\n').slice(0, 3).join('\n')
    };
  }
}

// Health check logic
export function assessProjectHealth(
  project: GrantProject,
  milestones: ProjectMilestone[],
  reports: ProjectReport[],
  budgetCategories: ProjectBudgetCategory[]
): { healthStatus: string; issues: string[]; recommendations: string[] } {
  const today = new Date();
  const issues: string[] = [];
  const recommendations: string[] = [];
  
  // Check overdue milestones
  const overdueMilestones = milestones.filter(m => 
    m.status !== 'completed' && m.status !== 'waived' &&
    new Date(m.dueDate) < today
  );
  if (overdueMilestones.length > 0) {
    issues.push(`${overdueMilestones.length} overdue milestone(s): ${overdueMilestones.map(m => m.title).join(', ')}`);
    recommendations.push('Update delayed milestones with new expected completion dates');
  }
  
  // Check upcoming milestone urgency (< 14 days)
  const urgentMilestones = milestones.filter(m => {
    const daysUntil = Math.ceil((new Date(m.dueDate).getTime() - today.getTime()) / 86400000);
    return m.status !== 'completed' && daysUntil >= 0 && daysUntil <= 14;
  });
  if (urgentMilestones.length > 0) {
    issues.push(`${urgentMilestones.length} milestone(s) due within 14 days`);
  }
  
  // Check overdue reports
  const overdueReports = reports.filter(r =>
    r.status !== 'submitted' && r.status !== 'approved' &&
    r.dueDate && new Date(r.dueDate) < today
  );
  if (overdueReports.length > 0) {
    issues.push(`${overdueReports.length} overdue report(s): ${overdueReports.map(r => r.title).join(', ')}`);
    recommendations.push('Submit overdue reports as soon as possible — contact funder if extension needed');
  }
  
  // Check budget burn rate
  if (project.projectStartDate && project.projectEndDate) {
    const totalDays = Math.ceil(
      (new Date(project.projectEndDate).getTime() - new Date(project.projectStartDate).getTime()) / 86400000
    );
    const elapsedDays = Math.ceil(
      (today.getTime() - new Date(project.projectStartDate).getTime()) / 86400000
    );
    const expectedBurnPercent = (elapsedDays / totalDays) * 100;
    
    const totalBudgeted = budgetCategories.reduce((sum, c) => sum + (c.budgetedAmountSek || 0), 0);
    const totalSpent = budgetCategories.reduce((sum, c) => sum + (c.spentAmountSek || 0), 0);
    const actualBurnPercent = totalBudgeted > 0 ? (totalSpent / totalBudgeted) * 100 : 0;
    
    if (actualBurnPercent > expectedBurnPercent * 1.25) {
      issues.push(`Budget burn rate is ${Math.round(actualBurnPercent)}% but project is only ${Math.round(expectedBurnPercent)}% through its timeline`);
      recommendations.push('Review spending — at current rate, budget may run out before project end');
    }
  }
  
  // Determine health status
  let healthStatus = 'on_track';
  if (overdueReports.length > 0 || overdueMilestones.length > 1) {
    healthStatus = 'delayed';
  } else if (overdueMilestones.length === 1 || urgentMilestones.length > 0 || issues.length > 0) {
    healthStatus = 'at_risk';
  }
  
  if (issues.length === 0) {
    recommendations.push('Project is on track — keep up the good work!');
  }
  
  return { healthStatus, issues, recommendations };
}

// Daily health check job
export const projectHealthJob = new CronJob('0 7 * * *', async () => {
  const activeProjects = await db.query.grantProjects.findMany({
    where: eq(grantProjects.status, 'active'),
    with: { milestones: true, reports: true, budgetCategories: true }
  });
  
  for (const project of activeProjects) {
    const { healthStatus, issues } = assessProjectHealth(
      project,
      project.milestones,
      project.reports,
      project.budgetCategories
    );
    
    // Update health status if changed
    if (healthStatus !== project.healthStatus) {
      await db.update(grantProjects)
        .set({ healthStatus, updatedAt: new Date() })
        .where(eq(grantProjects.id, project.id));
    }
    
    // Send notification emails for urgent issues
    if (issues.length > 0) {
      const user = await getUserById(project.userId);
      
      // Check if we already notified today (avoid spam)
      const lastNotified = await getLastNotificationDate(project.id, 'health_alert');
      const hoursSinceLast = lastNotified 
        ? (Date.now() - lastNotified.getTime()) / 3600000 
        : 999;
      
      if (hoursSinceLast > 23) {
        await sendProjectHealthEmail(user, project, issues);
        await recordNotification(project.id, 'health_alert');
      }
    }
  }
}, null, true, 'Europe/Stockholm');
