import { createServerClient } from '@supabase/ssr';
import { NextResponse, type NextRequest } from 'next/server';

export async function middleware(request: NextRequest) {
  let response = NextResponse.next({
    request: {
      headers: request.headers,
    },
  });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        get(name: string) {
          return request.cookies.get(name)?.value;
        },
        set(name: string, value: string, options: any) {
          request.cookies.set({
            name,
            value,
            ...options,
          });
          response = NextResponse.next({
            request: {
              headers: request.headers,
            },
          });
          response.cookies.set({
            name,
            value,
            ...options,
          });
        },
        remove(name: string, options: any) {
          request.cookies.set({
            name,
            value: '',
            ...options,
          });
          response = NextResponse.next({
            request: {
              headers: request.headers,
            },
          });
          response.cookies.set({
            name,
            value: '',
            ...options,
          });
        },
      },
    }
  );

  const {
    data: { user },
  } = await supabase.auth.getUser();

  // Protected routes - redirect to login if not authenticated
  const protectedPaths = ['/dashboard', '/bidrag', '/profile', '/applications'];
  const isProtectedPath = protectedPaths.some(path => 
    request.nextUrl.pathname.startsWith(path)
  );

  if (isProtectedPath && !user) {
    return NextResponse.redirect(new URL('/auth/login', request.url));
  }

  // If logged in and trying to access auth pages, redirect to dashboard
  const authPaths = ['/auth/login', '/auth/signup'];
  const isAuthPath = authPaths.some(path => 
    request.nextUrl.pathname.startsWith(path)
  );

  if (isAuthPath && user) {
    return NextResponse.redirect(new URL('/dashboard', request.url));
  }

  return response;
}

export const config = {
  matcher: [
    '/((?!_next/static|_next/image|favicon.ico|.*\\.(?:svg|png|jpg|jpeg|gif|webp)$).*)',
  ],
};

import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';

// Initialize Redis client
const redis = Redis.fromEnv();

// Different rate limiters for different endpoints
export const semanticMatchLimiter = new Ratelimit({
  redis,
  limiter: Ratelimit.slidingWindow(10, '1 h'), // 10 requests per hour
  analytics: true,
  prefix: 'ratelimit:semantic',
});

export const applicationGenerationLimiter = new Ratelimit({
  redis,
  limiter: Ratelimit.slidingWindow(20, '1 d'), // 20 generations per day
  analytics: true,
  prefix: 'ratelimit:generation',
});

export const apiLimiter = new Ratelimit({
  redis,
  limiter: Ratelimit.slidingWindow(100, '1 m'), // 100 general API calls per minute
  analytics: true,
  prefix: 'ratelimit:api',
});

// Helper function to check rate limit and return standardized response
export async function checkRateLimit(
  identifier: string,
  limiter: Ratelimit
) {
  const { success, limit, reset, remaining } = await limiter.limit(identifier);

  return {
    success,
    limit,
    remaining,
    reset: new Date(reset),
  };
}

// Helper to format rate limit error response
export function rateLimitError(remaining: number, reset: Date) {
  return Response.json(
    {
      error: 'Rate limit exceeded',
      message: `Du har använt alla dina AI-analyser. Försök igen ${reset.toLocaleTimeString('sv-SE')}.`,
      remaining,
      resetAt: reset.toISOString(),
    },
    { 
      status: 429,
      headers: {
        'X-RateLimit-Remaining': remaining.toString(),
        'X-RateLimit-Reset': reset.getTime().toString(),
      }
    }
  );
}

STEP 3.3: Update Notification Logic to Use Real User Data

Update app/api/cron/notifications/route.ts:
```typescript
import { createClient } from '@/lib/supabase/server';
import { sendNewGrantNotification, sendDeadlineReminder } from '@/lib/email/notifier';

export async function GET(request: Request) {
  // Verify cron secret
  const authHeader = request.headers.get('authorization');
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 });
  }
  
  const supabase = createClient();
  
  try {
    // 1. Find new grants added in last 24 hours
    const yesterday = new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString();
    
    const { data: newGrants } = await supabase
      .from('grants')
      .select('*')
      .gte('created_at', yesterday)
      .eq('status', 'open');
    
    // 2. Get all users with companies
    const { data: users } = await supabase
      .from('profiles')
      .select(`
        id,
        email,
        companies (
          id,
          company_name,
          industry,
          employees,
          revenue,
          location,
          description
        )
      `);
    
    let notificationsSent = 0;
    
    // 3. Send new grant notifications
    for (const grant of newGrants || []) {
      for (const user of users || []) {
        if (!user.companies || user.companies.length === 0) continue;
        
        for (const company of user.companies) {
          // Simple matching - you can make this more sophisticated
          const matchScore = calculateBasicMatchScore(company, grant);
          
          // Only notify if match > 60%
          if (matchScore >= 60 && user.email) {
            await sendNewGrantNotification(
              user.email,
              grant,
              company.company_name,
              matchScore
            );
            notificationsSent++;
            
            // Log notification sent
            await supabase.from('notifications').insert({
              user_id: user.id,
              grant_id: grant.id,
              type: 'new_grant_match',
              sent_at: new Date().toISOString()
            });
          }
        }
      }
    }
    
    // 4. Check for upcoming deadlines (7, 3, 1 day)
    const today = new Date();
    const deadlineChecks = [7, 3, 1];
    
    for (const daysAhead of deadlineChecks) {
      const targetDate = new Date(today);
      targetDate.setDate(targetDate.getDate() + daysAhead);
      
      const startOfDay = new Date(targetDate.setHours(0, 0, 0, 0)).toISOString();
      const endOfDay = new Date(targetDate.setHours(23, 59, 59, 999)).toISOString();
      
      const { data: upcomingGrants } = await supabase
        .from('grants')
        .select('*')
        .eq('status', 'open')
        .gte('deadline', startOfDay)
        .lte('deadline', endOfDay);
      
      for (const grant of upcomingGrants || []) {
        for (const user of users || []) {
          if (!user.companies || user.companies.length === 0) continue;
          
          for (const company of user.companies) {
            // Check if user has already been notified for this grant recently
            const { data: recentNotification } = await supabase
              .from('notifications')
              .select('id')
              .eq('user_id', user.id)
              .eq('grant_id', grant.id)
              .eq('type', 'deadline_reminder')
              .gte('sent_at', new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString())
              .single();
            
            if (!recentNotification && user.email) {
              await sendDeadlineReminder(
                user.email,
                grant,
                company.company_name
              );
              notificationsSent++;
              
              await supabase.from('notifications').insert({
                user_id: user.id,
                grant_id: grant.id,
                type: 'deadline_reminder',
                sent_at: new Date().toISOString()
              });
            }
          }
        }
      }
    }
    
    return Response.json({
      success: true,
      notificationsSent,
      newGrantsFound: newGrants?.length || 0
    });
    
  } catch (error: any) {
    console.error('Notification cron error:', error);
    return Response.json({
      success: false,
      error: error.message
    }, { status: 500 });
  }
}

function calculateBasicMatchScore(company: any, grant: any) {
  // Simplified matching - you already have better matching logic
  let score = 50; // Base score
  
  if (grant.keywords?.some((k: string) => 
    company.description?.toLowerCase().includes(k.toLowerCase())
  )) {
    score += 20;
  }
  
  if (grant.target_group?.includes(company.industry)) {
    score += 30;
  }
  
  return Math.min(100, score);
}
```

STEP 3.4: Add Notifications Table
```sql
CREATE TABLE IF NOT EXISTS notifications (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  grant_id UUID REFERENCES grants(id) ON DELETE CASCADE,
  type TEXT NOT NULL, -- 'new_grant_match', 'deadline_reminder', 'weekly_digest'
  sent_at TIMESTAMPTZ DEFAULT NOW(),
  opened_at TIMESTAMPTZ,
  clicked_at TIMESTAMPTZ
);

CREATE INDEX idx_notifications_user_id ON notifications(user_id);
CREATE INDEX idx_notifications_sent_at ON notifications(sent_at);

ALTER TABLE notifications ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can view own notifications"
  ON notifications FOR SELECT
  USING (auth.uid() = user_id);
```

STEP 4.2: Setup Cron-Job.org (FREE)

1. Go to https://cron-job.org/en/
2. Sign up for free account
3. Create three cron jobs:

**Job 1: Daily Scraping**
- URL: `https://your-domain.vercel.app/api/cron/scrape?frequency=daily`
- Schedule: Every day at 8:00 AM
- Request method: GET
- Headers: `Authorization: Bearer YOUR_CRON_SECRET`

**Job 2: Daily Notifications**
- URL: `https://your-domain.vercel.app/api/cron/notifications`
- Schedule: Every day at 9:00 AM (after scraping)
- Request method: GET
- Headers: `Authorization: Bearer YOUR_CRON_SECRET`

**Job 3: Weekly Digest**
- URL: `https://your-domain.vercel.app/api/cron/weekly-digest`
- Schedule: Every Monday at 8:00 AM
- Request method: GET
- Headers: `Authorization: Bearer YOUR_CRON_SECRET`

STEP 4.3: Alternative - Vercel Cron (Requires Pro Plan)

import { createClient } from '@/lib/supabase/server';

export default async function CronStatusPage() {
  const supabase = createClient();
  
  // Get recent scraper logs
  const { data: scraperLogs } = await supabase
    .from('scraper_logs')
    .select('*')
    .order('scraped_at', { ascending: false })
    .limit(10);
  
  // Get recent notifications
  const { data: recentNotifications } = await supabase
    .from('notifications')
    .select('*, profiles(email)')
    .order('sent_at', { ascending: false })
    .limit(20);
  
  return (
    <div className="container mx-auto p-6">
      <h1 className="text-2xl font-bold mb-6">Cron Job Status</h1>
      
      <div className="space-y-6">
        <div>

          <div className="space-y-2">
            {scraperLogs?.map(log => (
              <div key={log.id} className="p-4 border rounded">
                <div className="flex justify-between">
                  <span>{log.status === 'success' ? '✓' : '✗'} {log.status}</span>
                  <span>{new Date(log.scraped_at).toLocaleString('sv-SE')}</span>
                </div>
                <div className="text-sm text-muted-foreground">
                  Grants found: {log.grants_found}
                </div>
              </div>
            ))}
          </div>
        </div>
        
        <div>
          <h2 className="text-xl font-semibold mb-4">Recent Notifications</h2>
          <div className="space-y-2">
            {recentNotifications?.map(notif => (
              <div key={notif.id} className="p-4 border rounded">
                <div className="flex justify-between">
                  <span>{notif.type}</span>
                  <span>{new Date(notif.sent_at).toLocaleString('sv-SE')}</span>

                  To: {notif.profiles?.email}
                </div>
              </div>
            ))}

        </div>
      </div>
    </div>
  );
}
