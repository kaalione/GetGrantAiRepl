Build a complete, production-ready Success Fee pricing module for GetGrant.ai.
This gives free-plan users an alternative to subscription: pay nothing upfront,
pay 3% of the approved grant amount only when they win. Full Stripe billing,
admin oversight, legal terms, and upgrade conversion flow included.

import Anthropic from '@anthropic-ai/sdk';
import Stripe from 'stripe';

// ─── Fee calculation ───────────────────────────────────────────────────────

export interface FeeCalculation {
  approvedAmountSek: number;
  feePercentage: number;
  rawFeeSek: number;
  finalFeeSek: number;
  capApplied: boolean;
  minimumApplied: boolean;
  breakdown: string;              // human-readable explanation
}

export function calculateSuccessFee(
  approvedAmountSek: number,
  feePercentage: number = 3,
  maxCapSek: number = 25000,
  minFeeSek: number = 500
): FeeCalculation {
  const rawFee = Math.round(approvedAmountSek * (feePercentage / 100));
  const cappedFee = Math.min(rawFee, maxCapSek);
  const finalFee = Math.max(cappedFee, minFeeSek);
  
  const capApplied = rawFee > maxCapSek;
  const minimumApplied = cappedFee < minFeeSek;
  
  let breakdown = `${feePercentage}% × ${approvedAmountSek.toLocaleString('sv-SE')} SEK = ${rawFee.toLocaleString('sv-SE')} SEK`;
  if (capApplied) breakdown += ` → capped at ${maxCapSek.toLocaleString('sv-SE')} SEK`;
  if (minimumApplied) breakdown += ` → minimum fee ${minFeeSek.toLocaleString('sv-SE')} SEK applied`;
  
  return {
    approvedAmountSek,
    feePercentage,
    rawFeeSek: rawFee,
    finalFeeSek: finalFee,
    capApplied,
    minimumApplied,
    breakdown
  };
}

// Example table for documentation/UI:
// 50,000 SEK grant   × 3% = 1,500 SEK fee
// 100,000 SEK grant  × 3% = 3,000 SEK fee
// 500,000 SEK grant  × 3% = 15,000 SEK fee
// 1,000,000 SEK grant × 3% = 25,000 SEK (capped)
// 5,000,000 SEK grant × 3% = 25,000 SEK (capped)
// 10,000 SEK grant   × 3% = 500 SEK (minimum)

// ─── Stripe invoice creation ───────────────────────────────────────────────

export async function createFeeInvoice(
  agreement: SuccessFeeAgreement,
  feeCalc: FeeCalculation,
  user: { email: string; name: string; stripeCustomerId?: string },
  settings: SuccessFeeSettings
): Promise<{ invoiceId: string; invoiceUrl: string; hostedUrl: string }> {
  const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);
  
  // Create or retrieve Stripe customer
  let customerId = user.stripeCustomerId;
  if (!customerId) {
    const customer = await stripe.customers.create({
      email: user.email,
      name: user.name,
      metadata: { userId: agreement.userId, platform: 'getgrant_ai' }
    });
    customerId = customer.id;
    // Save customerId back to user record
    await db.update(users)
      .set({ stripeCustomerId: customerId })
      .where(eq(users.id, agreement.userId));
  }
  
  // Build detailed line item description
  const lineItemDescription = [
    `GetGrant.ai Success Fee`,
    `Grant: ${agreement.grantTitle}`,
    `Funder: ${agreement.funder}`,
    `Approved amount: ${feeCalc.approvedAmountSek.toLocaleString('sv-SE')} SEK`,
    `Fee rate: ${feeCalc.feePercentage}%`,
    feeCalc.capApplied ? `Maximum fee cap applied (${agreement.maxFeeCapSek!.toLocaleString('sv-SE')} SEK)` : '',
    feeCalc.minimumApplied ? `Minimum fee applied` : '',
    `Reference: ${agreement.grantAgreementRef || 'Not provided'}`
  ].filter(Boolean).join('\n');
  
  // Create invoice with auto-advance disabled so we can add items first
  const invoice = await stripe.invoices.create({
    customer: customerId,
    collection_method: 'send_invoice',
    days_until_due: settings.invoiceDaysUntilDue || 30,
    auto_advance: false,
    description: `GetGrant.ai success fee — ${agreement.grantTitle}`,
    footer: 'GetGrant.ai | support@getgrant.ai | VAT included where applicable',
    metadata: {
      agreementId: agreement.id,
      applicationId: agreement.applicationId,
      grantId: agreement.grantId,
      userId: agreement.userId,
      feeCalculation: JSON.stringify(feeCalc)
    },
    custom_fields: [
      { name: 'Bidragsreferens', value: agreement.grantAgreementRef || 'Ej angiven' },
      { name: 'Finansiär', value: agreement.funder },
      { name: 'GetGrant.ai referens', value: agreement.id.substring(0, 8).toUpperCase() }
    ]
  });
  
  // Add the fee line item
  await stripe.invoiceItems.create({
    customer: customerId,
    invoice: invoice.id,
    amount: feeCalc.finalFeeSek * 100,    // Stripe uses öre (1/100 SEK)
    currency: 'sek',
    description: lineItemDescription,
    tax_rates: [],                          // Add Swedish VAT tax rate ID if applicable
    metadata: {
      agreementId: agreement.id,
      feeBreakdown: feeCalc.breakdown
    }
  });
  
  // Finalize and send the invoice
  const finalizedInvoice = await stripe.invoices.finalizeInvoice(invoice.id);
  await stripe.invoices.sendInvoice(invoice.id);
  
  return {
    invoiceId: invoice.id,
    invoiceUrl: finalizedInvoice.invoice_pdf || '',
    hostedUrl: finalizedInvoice.hosted_invoice_url || ''
  };
}

// ─── Void an invoice (if agreement cancelled after invoicing) ─────────────

export async function voidFeeInvoice(stripeInvoiceId: string): Promise<void> {
  const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);
  await stripe.invoices.voidInvoice(stripeInvoiceId);
}

// ─── Upgrade comparison calculator ────────────────────────────────────────

export function calculateUpgradeComparison(
  totalFeesPaidSek: number,
  proMonthlySek: number = 495
): {
  annualSubscriptionSek: number;
  feesPaidSek: number;
  savingsSek: number;
  shouldShowUpgradePrompt: boolean;
  message: string;
} {
  const annualSek = proMonthlySek * 12;
  const savings = totalFeesPaidSek - annualSek;
  
  return {
    annualSubscriptionSek: annualSek,
    feesPaidSek: totalFeesPaidSek,
    savingsSek: savings,
    shouldShowUpgradePrompt: totalFeesPaidSek >= annualSek * 0.7, // show at 70% of annual cost
    message: savings > 0
      ? `Du har betalat ${totalFeesPaidSek.toLocaleString('sv-SE')} SEK i framgångsavgifter — ${savings.toLocaleString('sv-SE')} SEK mer än ett årsabonnemang kostar.`
      : `Du har betalat ${totalFeesPaidSek.toLocaleString('sv-SE')} SEK i framgångsavgifter. Ett årsabonnemang kostar bara ${annualSek.toLocaleString('sv-SE')} SEK.`
  };
}

// ─── Auto-expire old agreements ───────────────────────────────────────────

export async function expireStaleAgreements(autoExpireMonths: number = 18): Promise<number> {
  const cutoff = new Date();
  cutoff.setMonth(cutoff.getMonth() - autoExpireMonths);
  
  const stale = await db.update(successFeeAgreements)
    .set({ status: 'expired', updatedAt: new Date() })
    .where(
      and(
        eq(successFeeAgreements.status, 'active'),
        lt(successFeeAgreements.agreedAt, cutoff)
      )
    )
    .returning();
  
  // Log each expiration
  for (const agreement of stale) {
    await logFeeEvent(agreement.id, agreement.userId, 'agreement_expired',
      `Agreement expired after ${autoExpireMonths} months with no outcome reported`,
      { expiredAt: new Date().toISOString() },
      'system'
    );
  }
  
  return stale.length;
}

// ─── Event logging helper ─────────────────────────────────────────────────

export async function logFeeEvent(
  agreementId: string,
  userId: string,
  eventType: string,
  description: string,
  metadata: Record<string, unknown> = {},
  performedBy: string = 'system'
): Promise<void> {
  await db.insert(successFeeEvents).values({
    agreementId,
    userId,
    eventType,
    description,
    metadata,
    performedBy
  });
}

// User reports the outcome of their grant application.

Request body: {
  outcome: 'won' | 'rejected',
  approvedAmountSek?: number,     // required if outcome === 'won'
  grantAgreementRef?: string,     // optional — funder reference number
  proofOfApprovalUrl?: string,    // optional — link to decision letter
  rejectionReason?: string,       // optional if outcome === 'rejected'
  notes?: string
}

Validation:
- Agreement must belong to the user
- Status must be 'active'
- If outcome === 'won': approvedAmountSek required, must be > 0

Process if outcome === 'won':
1. Calculate fee using locked-in terms from agreement
2. Create Stripe invoice
3. Update agreement: status='grant_won' → 'fee_invoiced', store fee calculation
4. Log event: 'outcome_reported_won' + 'invoice_created'
5. Send confirmation email with invoice link
6. Check if upgrade prompt should be shown (calculateUpgradeComparison)
7. If upgrade prompt triggered, create successFeeUpgradePrompts record

Process if outcome === 'rejected':
1. Update agreement: status='rejected', outcomeReportedAt=now()
2. Log event: 'outcome_reported_rejected'
3. Send brief "Sorry to hear that" email with encouragement

Response (won): {
  status: 'fee_invoiced',
  feeCalculation: FeeCalculation,
  stripeInvoiceUrl: string,       // hosted invoice link
  dueDate: string,
  showUpgradePrompt: boolean,
  upgradeComparison?: object
}

Response (rejected): {
  status: 'rejected',
  message: 'No fee charged. Better luck next time!'
}

case 'invoice.paid': {
  const invoice = event.data.object as Stripe.Invoice;

  // Send payment confirmation email
  const agreement = await db.query.successFeeAgreements.findFirst({

    with: { user: true }
  });
  if (agreement) {

}

case 'invoice.payment_failed': {
  const invoice = event.data.object as Stripe.Invoice;
  const agreementId = invoice.metadata?.agreementId;
  if (!agreementId) break;
  
  await db.update(successFeeAgreements)
    .set({
      stripePaymentStatus: 'payment_failed',
      updatedAt: new Date()
    })
    .where(eq(successFeeAgreements.id, agreementId));
  
  await logFeeEvent(
    agreementId,
    invoice.metadata.userId,
    'payment_failed',
    'Invoice payment failed',
    { invoiceId: invoice.id, attemptCount: invoice.attempt_count },
    'stripe_webhook'
  );
  
  // Send payment failure email with link to invoice
  const agreement = await db.query.successFeeAgreements.findFirst({
    where: eq(successFeeAgreements.id, agreementId),
    with: { user: true }
  });
  if (agreement) {
    await sendPaymentFailedEmail(agreement, invoice.hosted_invoice_url || '');
  }
  break;
}

case 'invoice.voided': {
  const invoice = event.data.object as Stripe.Invoice;
  const agreementId = invoice.metadata?.agreementId;
  if (!agreementId) break;
  
  await db.update(successFeeAgreements)
    .set({ stripePaymentStatus: 'void', updatedAt: new Date() })
    .where(eq(successFeeAgreements.id, agreementId));
  
  await logFeeEvent(
    agreementId, invoice.metadata.userId, 'invoice_voided',
    'Invoice voided', { invoiceId: invoice.id }, 'stripe_webhook'
  );
  break;
}

import { CronJob } from 'cron';

// Daily at 09:00 — payment reminders for overdue/near-due invoices
export const feeReminderJob = new CronJob('0 9 * * *', async () => {
  const settings = await getSuccessFeeSettings();
  
  // Find agreements with invoices due within 7 days OR already overdue
  const agreements = await db.query.successFeeAgreements.findMany({
    where: and(
      eq(successFeeAgreements.status, 'fee_invoiced'),
      or(
        // Due within 7 days
        lte(successFeeAgreements.invoiceDueDate, 
            new Date(Date.now() + 7 * 24 * 60 * 60 * 1000).toISOString().split('T')[0]),
        // Already overdue
        lt(successFeeAgreements.invoiceDueDate, 
           new Date().toISOString().split('T')[0])
      )
    ),
    with: { user: true }
  });
  
  for (const agreement of agreements) {
    // Don't spam — max 1 reminder per 7 days
    const daysSinceLastReminder = agreement.reminderSentAt
      ? Math.floor((Date.now() - new Date(agreement.reminderSentAt).getTime()) / 86400000)
      : 999;
    
    if (daysSinceLastReminder >= 7) {
      const daysUntilDue = Math.ceil(
        (new Date(agreement.invoiceDueDate!).getTime() - Date.now()) / 86400000
      );
      
      await sendPaymentReminderEmail(agreement, daysUntilDue);
      
      await db.update(successFeeAgreements)
        .set({
          reminderSentAt: new Date(),
          reminderCount: (agreement.reminderCount || 0) + 1,
          updatedAt: new Date()
        })
        .where(eq(successFeeAgreements.id, agreement.id));
      
      await logFeeEvent(
        agreement.id, agreement.userId, 'reminder_sent',
        `Payment reminder sent (${daysUntilDue} days until due)`,
        { reminderCount: (agreement.reminderCount || 0) + 1 },
        'system'
      );
    }
  }
}, null, true, 'Europe/Stockholm');

// Weekly on Monday at 08:00 — expire stale agreements
export const feeExpirationJob = new CronJob('0 8 * * 1', async () => {
  const settings = await getSuccessFeeSettings();
  const expired = await expireStaleAgreements(settings.autoExpireMonths);
  if (expired > 0) {
    console.log(`Expired ${expired} stale success fee agreements`);
  }
}, null, true, 'Europe/Stockholm');

// Weekly on Monday at 08:30 — send upgrade prompts for eligible users
export const upgradePromptJob = new CronJob('30 8 * * 1', async () => {
  // Find users with unprompted upgrade opportunities
  const unprompted = await db.query.successFeeUpgradePrompts.findMany({
    where: and(
      eq(successFeeUpgradePrompts.promptShownAt, null), // never shown via email
      eq(successFeeUpgradePrompts.userClickedUpgrade, false)
    ),
    limit: 50  // batch limit
  });
  
  for (const prompt of unprompted) {
    const user = await getUserById(prompt.userId);
    if (user && user.plan === 'free') {  // only send if still on free plan
      await sendUpgradePromptEmail(user, prompt);
      await db.update(successFeeUpgradePrompts)
        .set({ promptShownAt: new Date() })
        .where(eq(successFeeUpgradePrompts.id, prompt.id));
    }
  }
}, null, true, 'Europe/Stockholm');
