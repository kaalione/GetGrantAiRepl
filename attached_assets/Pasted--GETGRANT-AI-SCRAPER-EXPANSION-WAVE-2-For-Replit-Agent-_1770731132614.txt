APPROACH — API-BASED (NOT web scraping):

The portal at https://ec.europa.eu/info/funding-tenders/opportunities/portal/screen/opportunities/topic-search
makes XHR/fetch calls to an internal API. You need to:

1. Use browser DevTools on the portal to identify the actual API endpoint
   The portal frontend makes POST requests to a search API endpoint.
   
2. The API endpoint pattern is typically:
   https://ec.europa.eu/info/funding-tenders/opportunities/data/topicSearch
   
   With a JSON POST body containing search filters like:
   {
     "callStatus": ["OPEN", "FORTHCOMING"],
     "frameworkProgramme": ["43108390"],  // Horizon Europe
     "sortBy": "deadlineDate",
     "orderBy": "asc",
     "pageSize": 100,
     "pageNumber": 1
   }

3. ALTERNATIVE APPROACH — The portal also publishes a static data file:
   https://ec.europa.eu/info/funding-tenders/opportunities/data/topic-list.html
   or similar data endpoints.
   
   There is also an open-source reference project that successfully fetches
   Horizon Europe calls: github.com/thisisjohnc/horizon-europe-tools
   This project uses the portal's public data to build spreadsheets of calls.
   Study its approach for the correct API endpoint and data format.

4. If the JSON API approach proves difficult, FALL BACK to web scraping the
   topic search results page with Playwright.

```python
# scrapers/sources/eu_funding_tenders.py

from sources.base_scraper import BaseScraper
import requests
import json
import time

class EUFundingTendersScraper(BaseScraper):
    """
    Scraper for EU Funding & Tenders Portal
    Fetches open and forthcoming calls from Horizon Europe, EIC,
    Digital Europe, LIFE, and other EU programs.
    
    Uses the portal's underlying search API (JSON).
    """
    
    def __init__(self, source_id=None):
        super().__init__(
            source_name="EU Funding & Tenders",
            base_url="https://ec.europa.eu/info/funding-tenders/opportunities/portal/",
            source_type="eu",
            default_category="eu",
            source_id=source_id
        )
        # The actual API endpoint — this needs to be verified by inspecting
        # the portal's network requests. Common patterns:
        self.api_url = "https://ec.europa.eu/info/funding-tenders/opportunities/data/topicSearch"
        
        # EU programs relevant for Swedish businesses
        self.relevant_programmes = [
            "Horizon Europe",
            "EIC Accelerator", 
            "Digital Europe",
            "LIFE",
            "Creative Europe",
            "Innovation Fund",
            "Interreg",
        ]
    
    def fetch_grants(self):
        """
        Fetch open and forthcoming calls from the EU F&T API.
        
        IMPORTANT: The exact API format needs to be determined by
        inspecting the portal's network requests. The structure below
        is a starting template — adapt based on actual API response.
        """
        all_grants = []
        
        for status in ["OPEN", "FORTHCOMING"]:
            try:
                # Attempt 1: JSON API
                grants = self._fetch_via_api(status)
                all_grants.extend(grants)
            except Exception as e:
                print(f"API approach failed for status {status}: {e}")
                print("Falling back to web scraping...")
                try:
                    # Attempt 2: Web scraping fallback
                    grants = self._fetch_via_scraping(status)
                    all_grants.extend(grants)
                except Exception as e2:
                    print(f"Web scraping also failed: {e2}")
        
        return all_grants
    
    def _fetch_via_api(self, status):
        """Try fetching via the portal's JSON API"""
        grants = []
        page = 1
        
        while True:
            # Construct the API request
            # NOTE: Adapt this payload based on actual API inspection
            payload = {
                "callStatus": [status],
                "sortBy": "deadlineDate",
                "orderBy": "asc", 
                "pageSize": 50,
                "pageNumber": page
            }
            
            headers = {
                "Content-Type": "application/json",
                "Accept": "application/json",
                "User-Agent": "GetGrant.ai Bot/1.0 (grant aggregator for Swedish businesses)"
            }
            
            response = requests.post(
                self.api_url,
                json=payload,
                headers=headers,
                timeout=30
            )
            response.raise_for_status()
            data = response.json()
            
            # Extract results — adapt field names based on actual API response
            results = data.get("topics", data.get("results", data.get("data", [])))
            
            if not results:
                break
            
            for item in results:
                raw = self._parse_api_result(item)
                if raw:
                    grants.append(raw)
            
            # Check if there are more pages
            total = data.get("totalCount", data.get("total", 0))
            if page * 50 >= total:
                break
            
            page += 1
            self.rate_limit()
        
        return grants
    
    def _parse_api_result(self, item):
        """Parse a single API result into raw grant data"""
        # Adapt field names based on actual API response structure
        return {
            "title": item.get("title", item.get("topicTitle", "")),
            "description": item.get("description", item.get("topicDescription", "")),
            "url": f"https://ec.europa.eu/info/funding-tenders/opportunities/portal/screen/opportunities/topic-details/{item.get('identifier', item.get('topicId', ''))}",
            "deadline_text": item.get("deadlineDate", item.get("deadline", "")),
            "status_text": item.get("status", item.get("callStatus", "")),
            "amount_text": item.get("budget", item.get("indicativeBudget", "")),
            "eligibility": item.get("eligibility", ""),
            "programme": item.get("programmeName", item.get("programme", "")),
            "category": "eu",
        }
    
    def _fetch_via_scraping(self, status):
        """Fallback: scrape the portal using Playwright"""
        url = f"https://ec.europa.eu/info/funding-tenders/opportunities/portal/screen/opportunities/topic-search;callStatus={status}"
        
        soup = self.fetch_with_playwright(url, wait_selector=".topic-list-item, .sedia-result-item, [class*='result']")
        if not soup:
            return []
        
        grants = []
        # Try common selector patterns for result items
        items = soup.select(".topic-list-item, .sedia-result-item, [class*='ResultItem'], .eui-card")
        
        for item in items:
            title_el = item.select_one("h3, h4, .title, [class*='title']")
            link_el = item.select_one("a[href*='topic-details']")
            deadline_el = item.select_one("[class*='deadline'], [class*='date'], time")
            status_el = item.select_one("[class*='status'], .badge, .tag")
            
            if title_el:
                grant_url = ""
                if link_el and link_el.get("href"):
                    href = link_el["href"]
                    if href.startswith("/"):
                        grant_url = f"https://ec.europa.eu{href}"
                    elif href.startswith("http"):
                        grant_url = href
                
                grants.append({
                    "title": title_el.get_text(strip=True),
                    "url": grant_url,
                    "description": "",
                    "deadline_text": deadline_el.get_text(strip=True) if deadline_el else "",
                    "status_text": status_el.get_text(strip=True) if status_el else status,
                    "amount_text": "",
                    "eligibility": "",
                    "category": "eu",
                })
        
        return grants
    
    def transform_to_grant(self, raw):
        """Transform EU F&T data to our grant schema"""
        grant = super().transform_to_grant(raw)
        
        # Override source-specific fields
        grant["source_name"] = "EU Funding & Tenders"
        grant["organization"] = raw.get("programme", "European Commission")
        grant["category"] = "eu"
        
        # Add EU-specific keywords
        eu_keywords = ["eu", "horizon europe", "european commission"]
        programme = raw.get("programme", "").lower()
        if "horizon" in programme:
            eu_keywords.append("horizon europe")
        if "eic" in programme:
            eu_keywords.extend(["eic", "startup", "deeptech"])
        if "digital" in programme:
            eu_keywords.extend(["digital", "ai", "cybersecurity"])
        if "life" in programme:
            eu_keywords.extend(["miljö", "klimat", "life"])
        
        existing_kw = grant.get("keywords", []) or []
        grant["keywords"] = list(set(existing_kw + eu_keywords))
        
        return grant
```

```python
from sources.base_scraper import BaseScraper
from utils.dedup import is_duplicate, title_similarity

class VerksamtScraper(BaseScraper):
    """
    Scraper for Verksamt.se — Sweden's official business support aggregator.
    
    IMPORTANT: This source aggregates grants from many agencies we already
    scrape individually (Vinnova, Tillväxtverket, Almi, etc.).
    Deduplication is critical to avoid inflating grant counts.
    """
    
    def __init__(self, source_id=None):
        super().__init__(
            source_name="Verksamt.se",
            base_url="https://www.verksamt.se/driva/finansiera-din-start/sok-finansiering",
            source_type="aggregator",
            default_category="general",
            source_id=source_id
        )
    
    def fetch_grants(self):
        """
        Scrape Verksamt.se search results using Playwright.
        The page uses JavaScript to load and filter results.
        """
        grants = []
        
        # Use Playwright since the page is JS-rendered
        soup = self.fetch_with_playwright(
            self.base_url,
            wait_selector=".search-result, .result-item, [class*='result'], .listing-item, article"
        )
        
        if not soup:
            print("Failed to load Verksamt.se — trying alternative selectors")
            # Try loading with a longer wait
            soup = self.fetch_with_playwright(
                self.base_url,
                wait_selector="body",
                timeout=15000
            )
            if not soup:
                raise Exception("Could not load Verksamt.se page")
        
        # Find all grant listing items
        # Verksamt.se may use various CSS classes — try multiple selectors
        items = soup.select(
            ".search-result-item, .result-item, .listing-item, "
            "[class*='SearchResult'], article.card, .card-body, "
            "li.list-item, .finansiering-item"
        )
        
        if not items:
            # If no items found with specific selectors, try broader approach
            # Look for links that point to financial support pages
            items = soup.select("a[href*='finansi'], a[href*='stod'], a[href*='bidrag']")
            
            for link in items:
                title = link.get_text(strip=True)
                href = link.get("href", "")
                
                if not title or len(title) < 5:
                    continue
                
                if href.startswith("/"):
                    href = f"https://www.verksamt.se{href}"
                
                grants.append({
                    "title": title,
                    "url": href,
                    "description": "",
                    "deadline_text": "",
                    "status_text": "",
                    "amount_text": "",
                    "eligibility": "",
                })
        else:
            for item in items:
                title_el = item.select_one("h2, h3, h4, .title, [class*='title'], a")
                link_el = item.select_one("a[href]")
                desc_el = item.select_one("p, .description, .summary, [class*='desc']")
                
                if not title_el:
                    continue
                
                title = title_el.get_text(strip=True)
                if not title or len(title) < 5:
                    continue
                
                url = ""
                if link_el and link_el.get("href"):
                    href = link_el["href"]
                    if href.startswith("/"):
                        url = f"https://www.verksamt.se{href}"
                    elif href.startswith("http"):
                        url = href
                
                grants.append({
                    "title": title,
                    "url": url,
                    "description": desc_el.get_text(strip=True) if desc_el else "",
                    "deadline_text": "",
                    "status_text": "",
                    "amount_text": "",
                    "eligibility": "",
                })
        
        # For each grant, try to fetch the detail page for more info
        enriched_grants = []
        for grant in grants[:80]:  # Limit to 80 to avoid too many requests
            if grant["url"]:
                self.rate_limit()
                try:
                    detail_soup = self.fetch_page(grant["url"])
                    if detail_soup:
                        # Extract more details from the detail page
                        desc = detail_soup.select_one(
                            ".body-text, .content, article, .main-content, "
                            "[class*='description'], .text-content"
                        )
                        if desc:
                            grant["description"] = desc.get_text(strip=True)[:2000]
                        
                        # Look for deadline info
                        deadline_el = detail_soup.find(
                            string=lambda t: t and any(w in t.lower() for w in 
                            ["sista dag", "deadline", "ansök senast", "sista ansökningsdag"])
                        )
                        if deadline_el:
                            grant["deadline_text"] = deadline_el.strip()
                        
                        # Look for amount info
                        amount_el = detail_soup.find(
                            string=lambda t: t and any(w in t.lower() for w in 
                            ["belopp", "kronor", "miljoner", "max", "upp till"])
                        )
                        if amount_el:
                            grant["amount_text"] = amount_el.strip()
                except Exception as e:
                    print(f"  Could not fetch detail for {grant['title']}: {e}")
            
            enriched_grants.append(grant)
        
        return enriched_grants
    
    def transform_to_grant(self, raw):
        """Transform Verksamt.se data to our schema"""
        grant = super().transform_to_grant(raw)
        grant["source_name"] = "Verksamt.se"
        grant["organization"] = "Verksamt.se (Aggregator)"
        return grant
```

Eufonder.se is a Swedish-language aggregator for EU funding opportunities.

Method: BeautifulSoup (static HTML)

```python
# scrapers/sources/eufonder.py

from sources.base_scraper import BaseScraper

class EuFonderScraper(BaseScraper):
    """
    Scraper for Eufonder.se — Swedish-language EU funding aggregator.
    
    Provides EU funding info in Swedish, complementing the EU F&T Portal.
    Will have overlap with EU F&T Portal — dedup handles this.
    """
    
    def __init__(self, source_id=None):
        super().__init__(
            source_name="Eufonder.se",
            base_url="https://eufonder.se/aktuellt/utlysningar.html",
            source_type="aggregator",
            default_category="eu",
            source_id=source_id
        )
    
    def fetch_grants(self):
        """Fetch EU grant listings from Eufonder.se"""
        grants = []
        
        # Fetch main utlysningar page
        soup = self.fetch_page(self.base_url)
        if not soup:
            # Try alternative URLs
            alt_urls = [
                "https://eufonder.se/aktuellt/utlysningar/",
                "https://eufonder.se/utlysningar.html",
                "https://eufonder.se/utlysningar/",
            ]
            for alt in alt_urls:
                soup = self.fetch_page(alt)
                if soup:
                    break
        
        if not soup:
            raise Exception("Could not load Eufonder.se")
        
        # Find grant listing items
        items = soup.select(
            "article, .article-item, .news-item, .utlysning, "
            ".listing-item, .content-item, .entry, "
            "li.item, .card, [class*='utlysning']"
        )
        
        if not items:
            # Broader fallback: find all internal links that look like grants
            items = soup.select("a[href*='utlys'], a[href*='fond'], a[href*='program']")
            
            for link in items:
                title = link.get_text(strip=True)
                href = link.get("href", "")
                
                if not title or len(title) < 10:
                    continue
                
                if href.startswith("/"):
                    href = f"https://eufonder.se{href}"
                elif not href.startswith("http"):
                    href = f"https://eufonder.se/{href}"
                
                grants.append({
                    "title": title,
                    "url": href,
                    "description": "",
                    "deadline_text": "",
                    "status_text": "",
                    "amount_text": "",
                    "eligibility": "",
                })
        else:
            for item in items:
                title_el = item.select_one("h2, h3, h4, .title, a")
                link_el = item.select_one("a[href]")
                desc_el = item.select_one("p, .summary, .excerpt, .description")
                date_el = item.select_one("time, .date, [class*='date']")
                
                if not title_el:
                    continue
                
                title = title_el.get_text(strip=True)
                if not title or len(title) < 5:
                    continue
                
                url = ""
                if link_el and link_el.get("href"):
                    href = link_el["href"]
                    if href.startswith("/"):
                        url = f"https://eufonder.se{href}"
                    elif href.startswith("http"):
                        url = href
                    else:
                        url = f"https://eufonder.se/{href}"
                
                grants.append({
                    "title": title,
                    "url": url,
                    "description": desc_el.get_text(strip=True) if desc_el else "",
                    "deadline_text": date_el.get_text(strip=True) if date_el else "",
                    "status_text": "",
                    "amount_text": "",
                    "eligibility": "",
                })
        
        # Fetch detail pages for grants missing descriptions
        for grant in grants[:60]:
            if not grant["description"] and grant["url"]:
                self.rate_limit()
                try:
                    detail = self.fetch_page(grant["url"])
                    if detail:
                        content = detail.select_one(
                            "article, .content, .article-content, "
                            ".entry-content, main, .text-content"
                        )
                        if content:
                            grant["description"] = content.get_text(strip=True)[:2000]
                except:
                    pass
        
        return grants
    
    def transform_to_grant(self, raw):
        """Transform Eufonder.se data to our schema"""
        grant = super().transform_to_grant(raw)
        grant["source_name"] = "Eufonder.se"
        grant["organization"] = "EU (via Eufonder.se)"
        grant["category"] = "eu"
        
        # Add EU keywords
        existing_kw = grant.get("keywords", []) or []
        grant["keywords"] = list(set(existing_kw + ["eu", "europeisk"]))
        
        return grant
```
