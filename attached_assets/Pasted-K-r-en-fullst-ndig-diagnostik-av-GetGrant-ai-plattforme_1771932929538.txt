## 1. DATABAS — GRANTS
Kör dessa SQL-queries och visa resultaten:
```sql
-- Total grants + fördelning per status
SELECT status, COUNT(*) as antal, ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER(), 1) as procent
FROM grants GROUP BY status ORDER BY antal DESC;

-- Grants per källa (source) med eligibility-täckning
SELECT 
  source,
  COUNT(*) as total,
  COUNT(CASE WHEN structured_eligibility IS NOT NULL THEN 1 END) as har_structured,
  COUNT(CASE WHEN eligibility_criteria IS NOT NULL AND eligibility_criteria != '' THEN 1 END) as har_raw,
  COUNT(CASE WHEN amount_min IS NOT NULL OR amount_max IS NOT NULL THEN 1 END) as har_belopp,
  COUNT(CASE WHEN deadline IS NOT NULL THEN 1 END) as har_deadline,
  ROUND(COUNT(CASE WHEN structured_eligibility IS NOT NULL THEN 1 END) * 100.0 / COUNT(*), 0) as eligibility_pct
FROM grants
GROUP BY source
ORDER BY total DESC;

-- Totala siffror
SELECT
  COUNT(*) as total_grants,
  COUNT(CASE WHEN structured_eligibility IS NOT NULL THEN 1 END) as structured_eligibility,
  ROUND(COUNT(CASE WHEN structured_eligibility IS NOT NULL THEN 1 END) * 100.0 / COUNT(*), 1) as structured_pct,
  COUNT(CASE WHEN status = 'open' THEN 1 END) as open_grants,
  COUNT(CASE WHEN status = 'upcoming' THEN 1 END) as upcoming_grants,
  COUNT(CASE WHEN amount_min IS NOT NULL OR amount_max IS NOT NULL THEN 1 END) as har_belopp,
  COUNT(CASE WHEN deadline IS NOT NULL THEN 1 END) as har_deadline,
  COUNT(DISTINCT source) as antal_kallor,
  COUNT(DISTINCT market) as antal_marknader
FROM grants;

-- Marknadsfördelning
SELECT market, COUNT(*) as antal FROM grants GROUP BY market ORDER BY antal DESC;
```
