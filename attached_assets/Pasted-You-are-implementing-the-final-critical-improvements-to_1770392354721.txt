STEP 1.1: Create Recently Viewed Hook

Create `src/hooks/useRecentlyViewed.ts`:
```typescript
import { useEffect, useState } from 'react';

interface RecentlyViewedGrant {
  id: string;
  title: string;
  sourceName: string;
  matchScore?: number;
  viewedAt: number;
}

const STORAGE_KEY = 'getgrant_recently_viewed';
const MAX_ITEMS = 5;

export function useRecentlyViewed() {
  const [recentlyViewed, setRecentlyViewed] = useState<RecentlyViewedGrant[]>([]);

  useEffect(() => {
    loadRecentlyViewed();
  }, []);

  function loadRecentlyViewed() {
    try {
      const stored = localStorage.getItem(STORAGE_KEY);
      if (stored) {
        const parsed = JSON.parse(stored);
        setRecentlyViewed(parsed);
      }
    } catch (error) {
      console.error('Failed to load recently viewed:', error);
    }
  }

  function addToRecentlyViewed(grant: {
    id: string;
    title: string;
    sourceName: string;
    matchScore?: number;
  }) {
    try {
      const existing = [...recentlyViewed];
      
      // Remove if already exists
      const filtered = existing.filter(g => g.id !== grant.id);
      
      // Add to start
      const updated = [
        {
          ...grant,
          viewedAt: Date.now(),
        },
        ...filtered,
      ].slice(0, MAX_ITEMS); // Keep only last 5

      setRecentlyViewed(updated);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(updated));
    } catch (error) {
      console.error('Failed to save recently viewed:', error);
    }
  }

  function clearRecentlyViewed() {
    setRecentlyViewed([]);
    localStorage.removeItem(STORAGE_KEY);
  }

  return {
    recentlyViewed,
    addToRecentlyViewed,
    clearRecentlyViewed,
  };
}
```

Create `server/routes/bookmarks.ts`:
```typescript
import express from 'express';
import { requireAuth } from '../middleware/auth';
import { db } from '../db';
import { grantBookmarks, grants } from '../db/schema';
import { eq, and, desc } from 'drizzle-orm';

const router = express.Router();

// Get user's bookmarks
router.get('/bookmarks', requireAuth, async (req, res) => {
  try {
    const bookmarks = await db.query.grantBookmarks.findMany({
      where: eq(grantBookmarks.userId, req.user.id),
      with: {
        grant: true,
      },
      orderBy: [desc(grantBookmarks.createdAt)],
    });

    res.json(bookmarks);
  } catch (error) {
    console.error('Get bookmarks error:', error);
    res.status(500).json({ error: 'Failed to fetch bookmarks' });
  }
});

// Check if grant is bookmarked
router.get('/bookmarks/check/:grantId', requireAuth, async (req, res) => {
  try {
    const bookmark = await db.query.grantBookmarks.findFirst({
      where: and(
        eq(grantBookmarks.userId, req.user.id),
        eq(grantBookmarks.grantId, req.params.grantId)
      ),
    });

    res.json({ bookmarked: !!bookmark });
  } catch (error) {
    console.error('Check bookmark error:', error);
    res.status(500).json({ error: 'Failed to check bookmark' });
  }
});

// Add bookmark
router.post('/bookmarks', requireAuth, async (req, res) => {
  try {
    const { grantId, notes } = req.body;

    if (!grantId) {
      return res.status(400).json({ error: 'Grant ID is required' });
    }

    // Check if already bookmarked
    const existing = await db.query.grantBookmarks.findFirst({
      where: and(
        eq(grantBookmarks.userId, req.user.id),
        eq(grantBookmarks.grantId, grantId)
      ),
    });

    if (existing) {
      return res.status(400).json({ error: 'Already bookmarked' });
    }

    // Create bookmark
    const [bookmark] = await db.insert(grantBookmarks)
      .values({
        userId: req.user.id,
        grantId,
        notes: notes || null,
      })
      .returning();

    res.json(bookmark);
  } catch (error) {
    console.error('Add bookmark error:', error);
    res.status(500).json({ error: 'Failed to add bookmark' });
  }
});

// Remove bookmark
router.delete('/bookmarks/:grantId', requireAuth, async (req, res) => {
  try {
    await db.delete(grantBookmarks)
      .where(
        and(
          eq(grantBookmarks.userId, req.user.id),
          eq(grantBookmarks.grantId, req.params.grantId)
        )
      );

    res.json({ success: true });
  } catch (error) {
    console.error('Remove bookmark error:', error);
    res.status(500).json({ error: 'Failed to remove bookmark' });
  }
});

// Update bookmark notes
router.patch('/bookmarks/:grantId', requireAuth, async (req, res) => {
  try {
    const { notes } = req.body;

    const [bookmark] = await db.update(grantBookmarks)
      .set({ notes })
      .where(
        and(
          eq(grantBookmarks.userId, req.user.id),
          eq(grantBookmarks.grantId, req.params.grantId)
        )
      )
      .returning();

    if (!bookmark) {
      return res.status(404).json({ error: 'Bookmark not found' });
    }

    res.json(bookmark);
  } catch (error) {
    console.error('Update bookmark error:', error);
    res.status(500).json({ error: 'Failed to update bookmark' });
  }
});

export default router;
```

```typescript
import { useState, useEffect } from 'react';
import { toast } from 'sonner';
import { useTranslation } from 'react-i18next';

export function useBookmark(grantId: string) {
  const { t } = useTranslation();
  const [bookmarked, setBookmarked] = useState(false);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    checkBookmark();
  }, [grantId]);

  async function checkBookmark() {
    try {
      const res = await fetch(`/api/bookmarks/check/${grantId}`);
      const data = await res.json();
      setBookmarked(data.bookmarked);
    } catch (error) {
      console.error('Failed to check bookmark:', error);
    } finally {
      setLoading(false);
    }
  }

  async function toggleBookmark() {
    try {
      if (bookmarked) {
        // Remove bookmark
        const res = await fetch(`/api/bookmarks/${grantId}`, {
          method: 'DELETE',
        });

        if (!res.ok) throw new Error('Failed to remove bookmark');

        setBookmarked(false);
        toast.success(t('bookmarks.removed'));
      } else {
        // Add bookmark
        const res = await fetch('/api/bookmarks', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ grantId }),
        });

        if (!res.ok) throw new Error('Failed to add bookmark');

        setBookmarked(true);
        toast.success(t('bookmarks.added'));
      }
    } catch (error) {
      console.error('Toggle bookmark error:', error);
      toast.error(t('bookmarks.error'));
    }
  }

  return {
    bookmarked,
    loading,
    toggleBookmark,
  };
}
```

```typescript
            title: 'TEST: AI-innovation för hållbar mobilitet',
            sourceName: 'Vinnova',
            deadline: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
            amountMin: 500000,
            amountMax: 2000000,
          }, 'Test AB', 85);
          break;

        case 'deadline-reminder':
          await sendDeadlineReminder(user.email, {
            title: 'TEST: Energieffektivisering i industrin',
            sourceName: 'Energimyndigheten',
            deadline: new Date(Date.now() + 3 * 24 * 60 * 60 * 1000),
          }, 'Test AB', 3);
          break;

        case 'weekly-digest':
          await sendWeeklyDigest(
            user.email,
            user.fullName || 'Test User',
            [
              { title: 'AI-bidrag test 1', matchScore: 90 },
              { title: 'Hållbarhet test 2', matchScore: 85 },
            ],
            [
              { title: 'Deadline test 1', deadline: new Date() },
            ]
          );
          break;

        case 'alert-match':
          await sendAlertNotification(
            user.email,
            'AI-bidrag över 500k',
            {
              title: 'TEST: Digitalisering av SME',
              sourceName: 'Tillväxtverket',
            },
            88
          );
          break;

        default:
          return res.status(400).json({ error: 'Unknown email type' });
      }

      res.json({ success: true, message: `Test email sent: ${type}` });
    } catch (error) {
      console.error('Test email error:', error);
      res.status(500).json({ error: 'Failed to send test email' });
    }
  });
}

export default router;
```
