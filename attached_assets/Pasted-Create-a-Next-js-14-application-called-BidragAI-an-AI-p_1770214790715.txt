DATABASE SCHEMA (Supabase):

1. grants table:
   - id (uuid, primary key)
   - title (text)
   - description (text)
   - source_name (text) - e.g., "Vinnova", "Tillväxtverket"
   - source_type (text) - "myndighet", "stiftelse", "eu"
   - url (text)
   - deadline (timestamp)
   - amount_min (numeric)
   - amount_max (numeric)
   - eligibility_criteria (jsonb)
   - target_group (text[]) - array: ["startup", "sme", "nonprofit"]
   - keywords (text[])
   - application_requirements (jsonb)
   - status (text) - "open", "upcoming", "closed"
   - created_at (timestamp)
   - updated_at (timestamp)
   - raw_data (jsonb) - original scraped data

2. companies table:
   - id (uuid, primary key)
   - user_id (uuid, foreign key to auth.users)
   - company_name (text)
   - org_number (text)
   - industry (text)
   - employees (integer)
   - revenue (numeric)
   - founded_year (integer)
   - description (text)
   - location (text)
   - created_at (timestamp)

3. applications table:
   - id (uuid, primary key)
   - company_id (uuid, foreign key)
   - grant_id (uuid, foreign key)
   - status (text) - "draft", "generated", "submitted"
   - generated_content (text)
   - ai_model_used (text)
   - match_score (numeric) - 0-100
   - created_at (timestamp)
   - submitted_at (timestamp)

4. scraper_sources table:
   - id (uuid, primary key)
   - name (text)
   - type (text) - "api", "scrape"
   - url (text)
   - scraper_type (text) - "playwright", "beautifulsoup", "api"
   - selectors (jsonb) - CSS selectors for scraping
   - active (boolean)
   - last_scraped (timestamp)
   - update_frequency (text) - "daily", "weekly"
   - api_config (jsonb) - for API sources

5. scraper_logs table:
   - id (uuid, primary key)
   - source_id (uuid, foreign key)
   - status (text) - "success", "failed"
   - grants_found (integer)
   - error_message (text)
   - scraped_at (timestamp)
