/**
 * Extract structured eligibility criteria from a single grant using Claude
 */
export async function extractEligibility(
  grantId: string,
  title: string, 
  description: string,
  eligibilityText: string,
  sourceName: string,
  amountDescription: string
): Promise<EligibilityCriteria | null> {
  
  // Build the text to analyze — combine all available info
  const textToAnalyze = [
    `BIDRAGSTITEL: ${title}`,
    `KÄLLA: ${sourceName}`,
    description ? `BESKRIVNING:\n${description}` : '',
    eligibilityText ? `BEHÖRIGHETSINFORMATION:\n${eligibilityText}` : '',
    amountDescription ? `BELOPPSINFO: ${amountDescription}` : '',
  ].filter(Boolean).join('\n\n');
  
  // Skip if there's very little text to analyze
  if (textToAnalyze.length < 50) {
    console.log(`  Skipping ${title} — too little text (${textToAnalyze.length} chars)`);
    return null;
  }
  
  try {
    const message = await client.messages.create({
      model: 'claude-sonnet-4-5-20250929',  // Use same model as rest of system
      max_tokens: 2000,
      messages: [
        {
          role: 'user',
          content: `${EXTRACTION_PROMPT}\n\n---\n\nTEXT ATT ANALYSERA:\n\n${textToAnalyze.substring(0, 4000)}`
          // Cap at 4000 chars to manage token usage
        }
      ],
    });
    
    const responseText = message.content[0].type === 'text' 
      ? message.content[0].text 
      : '';
    
    // Parse JSON from response
    // Handle potential markdown code blocks or extra whitespace
    let cleanJson = responseText.trim();
    cleanJson = cleanJson.replace(/^```json?\s*/i, '').replace(/\s*```$/i, '');
    
    const jsonMatch = cleanJson.match(/\{[\s\S]*\}/);
    if (!jsonMatch) {
      console.error(`  No JSON found in response for: ${title}`);
      return null;
    }
    
    const criteria: EligibilityCriteria = JSON.parse(jsonMatch[0]);
    
    // Add metadata
    criteria.extraction_date = new Date().toISOString();
    
    // Validate confidence score
    criteria.confidence_score = Math.min(1, Math.max(0, criteria.confidence_score || 0));
    
    return criteria;
    
  } catch (error) {
    console.error(`  Error extracting eligibility for "${title}":`, error);
    return null;
  }
}

/**
 * Process all grants that don't have eligibility_criteria yet.
 * Runs in batches with delays to respect API rate limits.
 */
export async function processAllGrants(options?: {
  batchSize?: number;
  delayBetweenBatchesMs?: number;
  onlyOpen?: boolean;
  forceReprocess?: boolean;
  maxGrants?: number;
}) {
  const {
    batchSize = 5,
    delayBetweenBatchesMs = 10000,  // 10 seconds between batches
    onlyOpen = false,
    forceReprocess = false,
    maxGrants = 500,
  } = options || {};
  
  console.log('='.repeat(60));
  console.log('AI ELIGIBILITY EXTRACTION');
  console.log(`Batch size: ${batchSize}, Delay: ${delayBetweenBatchesMs}ms`);
  console.log('='.repeat(60));
  
  // Build query to find grants needing processing
  let query = db.select().from(grants);
  
  if (!forceReprocess) {
    // Only process grants where eligibility_criteria is null or empty
    query = query.where(
      or(
        isNull(grants.eligibility_criteria),
        sql`${grants.eligibility_criteria}::text = 'null'`,
        sql`${grants.eligibility_criteria}::text = '{}'`,
        sql`${grants.eligibility_criteria}::text = '""'`
      )
    );
  }
  
  if (onlyOpen) {
    query = query.where(eq(grants.status, 'open'));
  }
  
  // Limit total grants to process
  const grantsToProcess = await query.limit(maxGrants);
  
  console.log(`Found ${grantsToProcess.length} grants to process`);
  
  let processed = 0;
  let successful = 0;
  let failed = 0;
  let skipped = 0;
  
  // Process in batches
  for (let i = 0; i < grantsToProcess.length; i += batchSize) {
    const batch = grantsToProcess.slice(i, i + batchSize);
    const batchNum = Math.floor(i / batchSize) + 1;
    const totalBatches = Math.ceil(grantsToProcess.length / batchSize);
    
    console.log(`\nBatch ${batchNum}/${totalBatches} (grants ${i + 1}-${i + batch.length})`);
    
    // Process batch items in parallel (within reason)
    const batchPromises = batch.map(async (grant) => {
      console.log(`  Processing: ${grant.title?.substring(0, 60)}...`);
      
      const criteria = await extractEligibility(
        grant.id,
        grant.title || '',
        grant.description || '',
        grant.eligibility || '',
        grant.source_name || grant.organization || '',
        grant.amount_description || ''
      );
      
      if (criteria) {
        // Update the grant in database
        try {
          await db.update(grants)
            .set({ 
              eligibility_criteria: criteria,
              // Also update target_groups if they were empty
              ...((!grant.target_groups || (Array.isArray(grant.target_groups) && grant.target_groups.length === 0)) && criteria.company_sizes?.size_category?.length > 0 ? {
                target_groups: criteria.company_sizes.size_category
              } : {})
            })
            .where(eq(grants.id, grant.id));
          
          console.log(`  ✓ Saved: ${grant.title?.substring(0, 50)} (confidence: ${criteria.confidence_score})`);
          successful++;
        } catch (dbError) {
          console.error(`  ✗ DB error for ${grant.title}: ${dbError}`);
          failed++;
        }
      } else {
        skipped++;
      }
      
      processed++;
    });
    
    await Promise.all(batchPromises);
    
    // Delay between batches to respect rate limits
    if (i + batchSize < grantsToProcess.length) {
      console.log(`  Waiting ${delayBetweenBatchesMs / 1000}s before next batch...`);
      await new Promise(resolve => setTimeout(resolve, delayBetweenBatchesMs));
    }
  }
  
  console.log('\n' + '='.repeat(60));
  console.log('EXTRACTION COMPLETE');
  console.log(`Total: ${processed}, Successful: ${successful}, Failed: ${failed}, Skipped: ${skipped}`);
  console.log('='.repeat(60));
  
  return { processed, successful, failed, skipped };
}

/**

export function calculateStructuredMatch(
  company: {
    industry: string;
    employees: number;
    revenue: number;  // in SEK
    location: string;
    founded_year: number;
    org_type: string;  // "aktiebolag", "enskild_firma", etc.
    description: string;
  },
  grant: {
    eligibility_criteria: EligibilityCriteria | null;
    keywords: string[];
    target_groups: string[];
    title: string;
    description: string;
  }
): { score: number; breakdown: MatchBreakdown; eligible: boolean; blockers: string[] } {
  
  const criteria = grant.eligibility_criteria;
  
  // If no structured criteria, fall back to basic matching
  if (!criteria || criteria.confidence_score < 0.2) {
    return fallbackBasicMatch(company, grant);
  }
  
  let totalScore = 0;
  let maxScore = 0;
  const breakdown: MatchBreakdown = {};
  const blockers: string[] = [];
  
  // ── COMPANY TYPE CHECK (Pass/Fail, 10 points) ──
  maxScore += 10;
  if (criteria.company_types.length === 0 || criteria.company_types.includes('all')) {
    totalScore += 10;
    breakdown.company_type = { score: 10, max: 10, note: 'Alla organisationstyper' };
  } else if (criteria.company_types.includes(company.org_type)) {
    totalScore += 10;
    breakdown.company_type = { score: 10, max: 10, note: `${company.org_type} är godkänd` };
  } else {
    breakdown.company_type = { score: 0, max: 10, note: `Kräver: ${criteria.company_types.join(', ')}` };
    blockers.push(`Organisationsform: Kräver ${criteria.company_types.join(' eller ')}`);
  }
  
  // ── COMPANY SIZE CHECK (25 points) ──
  maxScore += 25;
  const sizeOk = checkCompanySize(company.employees, criteria.company_sizes);
  if (sizeOk === 'match') {
    totalScore += 25;
    breakdown.company_size = { score: 25, max: 25, note: 'Företagsstorlek passar' };
  } else if (sizeOk === 'unknown') {
    totalScore += 15;  // Partial score if criteria not specified
    breakdown.company_size = { score: 15, max: 25, note: 'Storlekskrav ej specificerat' };
  } else {
    breakdown.company_size = { score: 0, max: 25, note: `Kräver ${criteria.company_sizes.description}` };
    blockers.push(`Företagsstorlek: ${criteria.company_sizes.description}`);
  }
  
  // ── GEOGRAPHY CHECK (20 points) ──
  maxScore += 20;
  const geoOk = checkGeography(company.location, criteria.geography);
  if (geoOk === 'match') {
    totalScore += 20;
    breakdown.geography = { score: 20, max: 20, note: 'Rätt geografiskt område' };
  } else if (geoOk === 'unknown') {
    totalScore += 12;
    breakdown.geography = { score: 12, max: 20, note: 'Geografikrav ej specificerat' };
  } else {
    breakdown.geography = { score: 0, max: 20, note: `Kräver: ${criteria.geography.description}` };
    blockers.push(`Geografi: ${criteria.geography.description}`);
  }
  
  // ── SECTOR/INDUSTRY MATCH (25 points) ──
  maxScore += 25;
  const sectorOk = checkSector(company.industry, criteria.sectors);
  if (sectorOk === 'match') {
    totalScore += 25;
    breakdown.sector = { score: 25, max: 25, note: 'Bransch matchar' };
  } else if (sectorOk === 'partial') {
    totalScore += 15;
    breakdown.sector = { score: 15, max: 25, note: 'Delvis branschmatchning' };
  } else if (sectorOk === 'unknown') {
    totalScore += 15;
    breakdown.sector = { score: 15, max: 25, note: 'Branschkrav ej specificerat' };
  } else {
    totalScore += 5;  // Small base score — might still be relevant
    breakdown.sector = { score: 5, max: 25, note: `Riktar sig till: ${criteria.sectors.join(', ')}` };
  }
  
  // ── COMPANY AGE CHECK (10 points) ──
  maxScore += 10;
  if (company.founded_year) {
    const companyAge = new Date().getFullYear() - company.founded_year;
    const ageOk = checkCompanyAge(companyAge, criteria.company_age);
    if (ageOk === 'match') {
      totalScore += 10;
      breakdown.company_age = { score: 10, max: 10, note: 'Företagsålder passar' };
    } else if (ageOk === 'unknown') {
      totalScore += 6;
      breakdown.company_age = { score: 6, max: 10, note: 'Ålderskrav ej specificerat' };
    } else {
      breakdown.company_age = { score: 0, max: 10, note: criteria.company_age.description };
      blockers.push(`Företagsålder: ${criteria.company_age.description}`);
    }
  } else {
    totalScore += 6;
    breakdown.company_age = { score: 6, max: 10, note: 'Företagets ålder okänd' };
  }
  
  // ── KEYWORD OVERLAP (10 points) ──
  maxScore += 10;
  const companyText = `${company.industry} ${company.description}`.toLowerCase();
  const matchedKeywords = (grant.keywords || []).filter(kw => 
    companyText.includes(kw.toLowerCase())
  );
  const keywordScore = Math.min(10, matchedKeywords.length * 2);
  totalScore += keywordScore;
  breakdown.keywords = { 
    score: keywordScore, max: 10, 
    note: matchedKeywords.length > 0 
      ? `Matchade nyckelord: ${matchedKeywords.join(', ')}`
      : 'Inga matchade nyckelord'
  };
  
  // Calculate final score (0-100)
  const finalScore = Math.round((totalScore / maxScore) * 100);
  
  // Determine if company is eligible (no hard blockers)
  const eligible = blockers.length === 0;
  
  return {
    score: finalScore,
    breakdown,
    eligible,
    blockers,
  };
}

// Helper functions
function checkCompanySize(employees: number, criteria: any): 'match' | 'no_match' | 'unknown' {
  if (!criteria || (!criteria.min_employees && !criteria.max_employees && criteria.size_category?.length === 0)) {
    return 'unknown';
  }
  if (criteria.max_employees && employees > criteria.max_employees) return 'no_match';
  if (criteria.min_employees && employees < criteria.min_employees) return 'no_match';
  return 'match';
}

function checkGeography(location: string, criteria: any): 'match' | 'no_match' | 'unknown' {
  if (!criteria || criteria.regions?.length === 0) return 'unknown';
  if (criteria.regions?.includes('hela_sverige')) return 'match';
  if (criteria.regions?.includes('eu') && criteria.eu_countries) return 'match';
  const locationLower = location?.toLowerCase() || '';
  if (criteria.regions?.some((r: string) => locationLower.includes(r))) return 'match';
  if (criteria.counties?.some((c: string) => locationLower.includes(c))) return 'match';
  return 'no_match';
}

function checkSector(industry: string, sectors: string[]): 'match' | 'partial' | 'no_match' | 'unknown' {
  if (!sectors || sectors.length === 0 || sectors.includes('all')) return 'unknown';
  const industryLower = industry?.toLowerCase() || '';
  if (sectors.some(s => industryLower.includes(s))) return 'match';
  // Check for related sectors
  const sectorRelations: Record<string, string[]> = {
    'tech': ['digital', 'ai', 'software', 'it'],
    'cleantech': ['energy', 'environment', 'climate'],
    'health': ['life_science', 'medtech', 'pharma'],
    'food': ['agriculture', 'agtech'],
  };
  for (const [sector, related] of Object.entries(sectorRelations)) {
    if (sectors.includes(sector) && related.some(r => industryLower.includes(r))) return 'partial';
    if (related.some(r => sectors.includes(r)) && industryLower.includes(sector)) return 'partial';
  }
  return 'no_match';
}

function checkCompanyAge(age: number, criteria: any): 'match' | 'no_match' | 'unknown' {
  if (!criteria || (!criteria.min_years && !criteria.max_years)) return 'unknown';
  if (criteria.max_years && age > criteria.max_years) return 'no_match';
  if (criteria.min_years && age < criteria.min_years) return 'no_match';
  return 'match';
}

interface MatchBreakdown {
  [key: string]: { score: number; max: number; note: string };
}
