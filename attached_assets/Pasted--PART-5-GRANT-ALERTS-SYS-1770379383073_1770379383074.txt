```typescript
// In your schema file

export const grantAlerts = pgTable('grant_alerts', {
  id: uuid('id').primaryKey().defaultRandom(),
  userId: uuid('user_id').references(() => users.id).notNull(),
  companyId: uuid('company_id').references(() => companies.id),
  
  // Alert configuration
  name: text('name').notNull(), // User-friendly name
  active: boolean('active').default(true),
  
  // Filter criteria (all optional - null means "any")
  keywords: jsonb('keywords'), // Array of keywords to match
  sources: jsonb('sources'), // Array of source names (e.g., ["Vinnova", "Tillväxtverket"])
  minAmount: numeric('min_amount', { precision: 12, scale: 2 }),
  maxAmount: numeric('max_amount', { precision: 12, scale: 2 }),
  industries: jsonb('industries'), // Array of industries
  minMatchScore: integer('min_match_score').default(60),
  
  // Notification settings
  notifyImmediately: boolean('notify_immediately').default(true),
  includeInDigest: boolean('include_in_digest').default(true),
  
  // Tracking
  lastTriggered: timestamp('last_triggered'),
  triggerCount: integer('trigger_count').default(0),
  
  createdAt: timestamp('created_at').defaultNow(),
  updatedAt: timestamp('updated_at').defaultNow(),
});

export const alertMatches = pgTable('alert_matches', {
  id: uuid('id').primaryKey().defaultRandom(),
  alertId: uuid('alert_id').references(() => grantAlerts.id).notNull(),
  grantId: uuid('grant_id').references(() => grants.id).notNull(),
  matchScore: integer('match_score'),
  notified: boolean('notified').default(false),
  notifiedAt: timestamp('notified_at'),
  createdAt: timestamp('created_at').defaultNow(),
});
```

```typescript
async function checkAlertAgainstExistingGrants(alert: any) {
  try {
    // Get all open grants
    const openGrants = await db.query.grants.findMany({
      where: eq(grants.status, 'open'),
    });

    // Get user's company if specified

    if (alert.companyId) {
      company = await db.query.companies.findFirst({

      });
    }

    // Check each grant against alert criteria
    for (const grant of openGrants) {
      if (grantMatchesAlert(grant, alert, company)) {
        // Calculate match score if company exists
        let matchScore = 50; // Default
        if (company) {
          const match = calculateBasicMatch(company, grant);
          matchScore = match.score;
        }

        if (matchScore >= alert.minMatchScore) {

          const existingMatch = await db.query.alertMatches.findFirst({

              grantId: grant.id,

}

function grantMatchesAlert(grant: any, alert: any, company: any): boolean {
  // Check keywords
  if (alert.keywords && alert.keywords.length > 0) {
    const grantText = `${grant.title} ${grant.description}`.toLowerCase();
    const hasKeyword = alert.keywords.some((keyword: string) =>
      grantText.includes(keyword.toLowerCase())
    );
    if (!hasKeyword) return false;
  }

  // Check sources
  if (alert.sources && alert.sources.length > 0) {
    if (!alert.sources.includes(grant.sourceName)) {
      return false;
    }
  }

  // Check amount range
  if (alert.minAmount && grant.amountMax) {
    if (parseFloat(grant.amountMax) < parseFloat(alert.minAmount)) {
      return false;
    }
  }

  if (alert.maxAmount && grant.amountMin) {
    if (parseFloat(grant.amountMin) > parseFloat(alert.maxAmount)) {
      return false;
    }
  }

  // Check industries (if company specified)
  if (alert.industries && alert.industries.length > 0 && company) {
    if (!alert.industries.includes(company.industry)) {
      return false;
    }
  }

  return true;
}
```

```typescript
async function checkNewGrantsAgainstAlerts() {
  // Get all active alerts
  const activeAlerts = await db.query.grantAlerts.findMany({
    where: eq(grantAlerts.active, true),
  });

  // Get grants created in last 24 hours
  const yesterday = new Date(Date.now() - 24 * 60 * 60 * 1000);
  const newGrants = await db.query.grants.findMany({
    where: and(
      eq(grants.status, 'open'),
      gte(grants.createdAt, yesterday)
    ),
  });

  // Check each alert against new grants
  for (const alert of activeAlerts) {
    let company = null;
    if (alert.companyId) {
      company = await db.query.companies.findFirst({
        where: eq(companies.id, alert.companyId),
      });
    }

    for (const grant of newGrants) {
      if (grantMatchesAlert(grant, alert, company)) {
        // Calculate match score
        let matchScore = 50;
        if (company) {
          const match = calculateBasicMatch(company, grant);
          matchScore = match.score;
        }

        if (matchScore >= alert.minMatchScore) {
          // Check if match already exists
          const existingMatch = await db.query.alertMatches.findFirst({
            where: and(
              eq(alertMatches.alertId, alert.id),
              eq(alertMatches.grantId, grant.id)
            ),
          });

          if (!existingMatch) {
            // Create match
            await db.insert(alertMatches).values({
              alertId: alert.id,
              grantId: grant.id,
              matchScore,
            });

            // Update alert trigger count
            await db.update(grantAlerts)
              .set({
                triggerCount: sql`${grantAlerts.triggerCount} + 1`,
                lastTriggered: new Date(),
              })
              .where(eq(grantAlerts.id, alert.id));

            // Send notification if enabled
            if (alert.notifyImmediately) {
              const user = await db.query.users.findFirst({
                where: eq(users.id, alert.userId),
              });

              if (user?.email) {
                await sendAlertNotification(
                  user.email,
                  alert.name,
                  grant,
                  matchScore
                );
              }
            }
          }
        }
      }
    }
  }
}
```

```typescript
export function GrantAlerts() {
  const { t } = useTranslation();
  const [alerts, setAlerts] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [dialogOpen, setDialogOpen] = useState(false);
  const [editingAlert, setEditingAlert] = useState<any>(null);

  useEffect(() => {
    fetchAlerts();
  }, []);

  async function fetchAlerts() {
    try {
      const res = await fetch('/api/alerts');
      const data = await res.json();
      setAlerts(data);
    } catch (error) {
      console.error('Failed to fetch alerts:', error);
      toast.error(t('alerts.fetchError'));
    } finally {
      setLoading(false);
    }
  }

  function openCreateDialog() {
    setEditingAlert(null);
    setDialogOpen(true);
  }

  function openEditDialog(alert: any) {
    setEditingAlert(alert);
    setDialogOpen(true);
  }

  async function handleDelete(alertId: string) {
    if (!confirm(t('alerts.deleteConfirm'))) return;

    try {
      const res = await fetch(`/api/alerts/${alertId}`, {
        method: 'DELETE',
      });

      if (!res.ok) throw new Error('Delete failed');

      toast.success(t('alerts.deleteSuccess'));
      fetchAlerts();
    } catch (error) {
      console.error('Failed to delete alert:', error);
      toast.error(t('alerts.deleteError'));
    }

  async function handleToggle(alertId: string, active: boolean) {
    try {
      const res = await fetch(`/api/alerts/${alertId}`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ active }),

      console.error('Failed to toggle alert:', error);
      toast.error(t('alerts.toggleError'));

  }

  if (loading) {
    return (
      <Card className="animate-pulse">
        <CardContent className="p-6">

function AlertDialog({ alert, onSave, onCancel }: any) {

  const [formData, setFormData] = useState({
    name: alert?.name || '',
    keywords: alert?.keywords?.join(', ') || '',
    sources: alert?.sources || [],
    minAmount: alert?.minAmount || '',
    maxAmount: alert?.maxAmount || '',
    minMatchScore: alert?.minMatchScore || 60,
    notifyImmediately: alert?.notifyImmediately !== false,
  });
  const [saving, setSaving] = useState(false);

  async function handleSave() {
    if (!formData.name.trim()) {
      toast.error(t('alerts.nameRequired'));
      return;
    }

    setSaving(true);
    try {
      const payload = {
        ...formData,
        keywords: formData.keywords
          ? formData.keywords.split(',').map(k => k.trim()).filter(Boolean)
          : null,
      };

      const url = alert ? `/api/alerts/${alert.id}` : '/api/alerts';
      const method = alert ? 'PATCH' : 'POST';

      const res = await fetch(url, {
        method,
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      });

      if (!res.ok) throw new Error('Save failed');

      toast.success(alert ? t('alerts.updated') : t('alerts.created'));
      onSave();
    } catch (error) {
      console.error('Failed to save alert:', error);
      toast.error(t('alerts.saveError'));
    } finally {
      setSaving(false);
    }
  }

  return (
    <>
      <DialogHeader>
        <DialogTitle>
          {alert ? t('alerts.edit') : t('alerts.create')}
        </DialogTitle>
        <DialogDescription>
          {t('alerts.dialogDescription')}
        </DialogDescription>
      </DialogHeader>

      <div className="space-y-4 py-4">
        <div>
          <Label htmlFor="name">{t('alerts.name')} *</Label>
          <Input
            id="name"
            value={formData.name}
            onChange={(e) => setFormData({ ...formData, name: e.target.value })}
            placeholder={t('alerts.namePlaceholder')}
          />
        </div>

            value={formData.keywords}

          <div>

              value={formData.maxAmount}

              placeholder="5000000"
            />
          </div>
        </div>

        <div>
          <Label htmlFor="minMatchScore">{t('alerts.minScore')} ({formData.minMatchScore}%)</Label>
          <input
            type="range"
            id="minMatchScore"
            min="40"
            max="100"
            step="5"
            value={formData.minMatchScore}
            onChange={(e) => setFormData({ ...formData, minMatchScore: parseInt(e.target.value) })}
            className="w-full"
          />
        </div>

        <div className="flex items-center justify-between">
          <Label htmlFor="notifyImmediately">{t('alerts.notifyImmediately')}</Label>
          <Switch
            id="notifyImmediately"
            checked={formData.notifyImmediately}
            onCheckedChange={(checked) => setFormData({ ...formData, notifyImmediately: checked })}
          />
        </div>
      </div>

      <div className="flex gap-3">
        <Button onClick={handleSave} disabled={saving} className="flex-1">
          {saving ? t('common.saving') : t('common.save')}
        </Button>
        <Button variant="outline" onClick={onCancel}>
          {t('common.cancel')}
        </Button>
      </div>
    </>
  );
}
```
