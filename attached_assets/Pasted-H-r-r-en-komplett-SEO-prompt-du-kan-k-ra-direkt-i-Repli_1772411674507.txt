PART 2 — React Helmet / Per-Page Meta Tags
Install react-helmet-async if not already installed:
npm install react-helmet-async
In client/src/main.tsx (or App.tsx), wrap the app with HelmetProvider:

import { HelmetProvider } from 'react-helmet-async';
// wrap <App /> with <HelmetProvider>


Create a reusable SEO component at client/src/components/SEO.tsx:

import { Helmet } from 'react-helmet-async';

interface SEOProps {
  title?: string;
  description?: string;
  canonical?: string;
  noindex?: boolean;
}

const SITE_NAME = 'getgrant.ai';
const BASE_URL = 'https://getgrant.ai';
const DEFAULT_DESC = 'AI-driven bidragsplattform för svenska företag. Hitta rätt bidrag och ansök på minuter.';

export function SEO({ title, description, canonical, noindex }: SEOProps) {
  const fullTitle = title ? `${title} | ${SITE\_NAME}` : `${SITE_NAME} – Hitta svenska bidrag med AI`;
  
  return (
    <Helmet>
      <title>{fullTitle}</title>
      <meta name="description" content={description || DEFAULT_DESC} />
      {canonical && <link rel="canonical" href={`${BASE\_URL}${canonical}`} />}
      {noindex && <meta name="robots" content="noindex, nofollow" />}
      <meta property="og:title" content={fullTitle} />
      <meta property="og:description" content={description || DEFAULT_DESC} />
    </Helmet>
  );
}
