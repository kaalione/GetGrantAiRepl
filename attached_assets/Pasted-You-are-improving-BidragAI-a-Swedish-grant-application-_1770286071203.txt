```python
import requests
import os
from datetime import datetime
from utils.db import upsert_grant, log_scrape_result

class GDPApiClient:
    """
    Client for Swedish GDP (Gemensamma Data Projektet) API
    Fetches grant opportunities from Vinnova, Forte, Formas, Vetenskapsrådet, Energimyndigheten
    """
    
    def __init__(self):
        self.base_url = "https://api.vinnova.se/gdp/v2"
        self.api_key = os.getenv('GDP_API_KEY')
        self.headers = {
            'Ocp-Apim-Subscription-Key': self.api_key,
            'Accept': 'application/json'
        }
    
    def fetch_utlysningar(self, limit=100):
        """
        Fetch open grant calls (utlysningar)
        Returns list of grants in GDP format
        """
        try:
            url = f"{self.base_url}/utlysningar"
            params = {
                'limit': limit,
                'status': 'open'  # Only fetch open calls
            }
            
            response = requests.get(url, headers=self.headers, params=params)
            response.raise_for_status()
            
            data = response.json()
            return data.get('items', [])
            
        except requests.exceptions.RequestException as e:
            print(f"Error fetching GDP data: {e}")
            return []
    
    def transform_to_grant(self, gdp_item):
        """
        Transform GDP format to BidragAI grant schema
        
        GDP fields we care about:
        - titel (title)
        - beskrivning (description)
        - finansiar (funding body: Vinnova, Forte, etc)
        - sistaSokdatum (deadline)
        - kategoriFinansiar (keywords/categories)
        - webbadress (URL)
        """
        
        # Parse deadline
        deadline = None
        if gdp_item.get('sistaSokdatum'):
            try:
                deadline = datetime.fromisoformat(gdp_item['sistaSokdatum'].replace('Z', '+00:00'))
            except:
                pass
        
        # Determine source name
        source_name = gdp_item.get('finansiar', {}).get('namn', 'Unknown')
        
        # Extract keywords from categories
        keywords = []
        if 'kategoriFinansiar' in gdp_item:
            categories = gdp_item.get('kategoriFinansiar', [])
            keywords = [cat.get('namn') for cat in categories if cat.get('namn')]
        
        # Build grant object matching our DB schema
        grant = {
            'title': gdp_item.get('titel', 'Untitled'),
            'description': gdp_item.get('beskrivning', ''),
            'source_name': source_name,
            'source_type': 'myndighet',
            'url': gdp_item.get('webbadress', ''),
            'deadline': deadline,
            'amount_min': None,  # GDP doesn't always provide amounts
            'amount_max': None,
            'eligibility_criteria': {
                'categories': keywords,
                'funding_body': source_name
            },
            'target_group': self._extract_target_groups(gdp_item),
            'keywords': keywords,
            'application_requirements': {},
            'status': 'open' if deadline and deadline > datetime.now() else 'closed',
            'raw_data': gdp_item  # Store original for reference
        }
        
        return grant
    
    def _extract_target_groups(self, gdp_item):
        """
        Try to determine target groups from GDP data
        Returns array like: ['startup', 'sme', 'research']
        """
        target_groups = []
        
        description = gdp_item.get('beskrivning', '').lower()
        title = gdp_item.get('titel', '').lower()
        combined = f"{title} {description}"
        
        # Pattern matching for common target groups
        if any(word in combined for word in ['startup', 'nystartade', 'nya företag']):
            target_groups.append('startup')
        
        if any(word in combined for word in ['små och medelstora', 'sme', 'små företag']):
            target_groups.append('sme')
        
        if any(word in combined for word in ['forskare', 'universitet', 'högskola', 'akademi']):
            target_groups.append('research')
        
        if any(word in combined for word in ['ideell', 'förening', 'organisation']):
            target_groups.append('nonprofit')
        
        # Default if nothing matched
        if not target_groups:
            target_groups.append('all')
        
        return target_groups
    
    def scrape(self, source_id):
        """
        Main scraping function called by scheduler
        """
        print(f"Starting GDP API scrape...")
        
        try:
            # Fetch grants from API
            gdp_items = self.fetch_utlysningar(limit=200)
            print(f"Fetched {len(gdp_items)} items from GDP API")
            
            grants_inserted = 0
            
            # Transform and insert each grant
            for item in gdp_items:
                grant = self.transform_to_grant(item)
                
                # Only insert if it has minimum required fields
                if grant['title'] and grant['url']:
                    upsert_grant(grant)
                    grants_inserted += 1
            
            # Log successful scrape
            log_scrape_result(
                source_id=source_id,
                status='success',
                grants_found=grants_inserted,
                error_message=None
            )
            
            print(f"Successfully inserted/updated {grants_inserted} grants")
            return grants_inserted
            
        except Exception as e:
            error_msg = str(e)
            print(f"GDP scrape failed: {error_msg}")
            
            log_scrape_result(
                source_id=source_id,
                status='failed',
                grants_found=0,
                error_message=error_msg
            )
            
            return 0

# Main execution
if __name__ == "__main__":
    client = GDPApiClient()
    result = client.scrape(source_id=None)  # Pass actual source_id in production
    print(f"Scrape complete: {result} grants")
```

```typescript
import Anthropic from '@anthropic-ai/sdk';

const client = new Anthropic({
  apiKey: process.env.ANTHROPIC_API_KEY,
});

interface SemanticMatchResult {
  score: number; // 0-100
  reasoning: string;
  strengths: string[];
  concerns: string[];
}

export async function calculateSemanticMatch(
  companyDescription: string,
  companyIndustry: string,
  grantTitle: string,
  grantDescription: string,
  grantKeywords: string[]
): Promise<SemanticMatchResult> {
  
  const prompt = `Du är en expert på svenska företagsbidrag. Analysera hur väl detta företag passar för detta bidrag.

FÖRETAG:
Bransch: ${companyIndustry}
Beskrivning: ${companyDescription}

BIDRAG:
Titel: ${grantTitle}
Beskrivning: ${grantDescription}
Nyckelord: ${grantKeywords.join(', ')}

Analysera matchningen och svara med ENDAST en JSON-struktur (inga andra tecken):
{
  "score": <number 0-100>,
  "reasoning": "<kort förklaring på svenska>",
  "strengths": ["<styrka 1>", "<styrka 2>"],
  "concerns": ["<oro 1>", "<oro 2>"]
}`;

  try {
    const message = await client.messages.create({
      model: 'claude-sonnet-4-5-20250929',
      max_tokens: 1000,
      messages: [
        {
          role: 'user',
          content: prompt
        }
      ],
    });
    
    const responseText = message.content[0].text;
    
    // Extract JSON from response (handle markdown code blocks)
    const jsonMatch = responseText.match(/\{[\s\S]*\}/);
    if (!jsonMatch) {
      throw new Error('No JSON found in response');
    }
    
    const result = JSON.parse(jsonMatch[0]);
    
    return {
      score: Math.min(100, Math.max(0, result.score)),
      reasoning: result.reasoning,
      strengths: result.strengths || [],
      concerns: result.concerns || []
    };
    
  } catch (error) {
    console.error('Semantic matching error:', error);
    // Fallback to basic matching
    return {
      score: 50,
      reasoning: 'Automatisk analys misslyckades',
      strengths: [],
      concerns: []
    };
  }
}
```

```typescript
import { createClient } from '@/lib/supabase/server';
import { calculateSemanticMatch } from '@/lib/matching/semantic';

export async function POST(request: Request) {
  const { grantId, companyId } = await request.json();
  
  const supabase = createClient();
  
  // Fetch grant and company
  const { data: grant } = await supabase
    .from('grants')
    .select('*')
    .eq('id', grantId)
    .single();
    
  const { data: company } = await supabase
    .from('companies')
    .select('*')
    .eq('id', companyId)
    .single();
  
  if (!grant || !company) {
    return Response.json({ error: 'Not found' }, { status: 404 });
  }
  
  // Calculate basic rule-based match
  const basicMatch = calculateBasicMatch(company, grant);
  
  // Calculate semantic match using AI
  const semanticMatch = await calculateSemanticMatch(
    company.description || '',
    company.industry || '',
    grant.title,
    grant.description || '',
    grant.keywords || []
  );
  
  // Weighted combination: 40% basic rules, 60% semantic
  const finalScore = Math.round(
    (basicMatch.score * 0.4) + (semanticMatch.score * 0.6)
  );
  
  return Response.json({
    score: finalScore,
    basic: basicMatch,
    semantic: semanticMatch,
    explanation: `Matchning baserad på företagsprofil (${basicMatch.score}%) och AI-analys av beskrivningar (${semanticMatch.score}%).`
  });
}

function calculateBasicMatch(company, grant) {
  let score = 0;
  const factors = [];
  
  // Industry match
  if (grant.target_group?.includes(company.industry)) {
    score += 25;
    factors.push('Bransch matchar');
  }
  
  // Size match
  if (company.employees >= 1 && company.employees <= 50) {
    score += 15;
    factors.push('Företagsstorlek passar');
  }
  
  // Location match (if grant has regional requirements)
  if (grant.eligibility_criteria?.regions?.includes(company.location)) {
    score += 20;
    factors.push('Geografisk matchning');
  }
  
  // Keywords overlap
  if (company.description && grant.keywords) {
    const companyWords = company.description.toLowerCase().split(/\s+/);
    const matches = grant.keywords.filter(k => 
      companyWords.includes(k.toLowerCase())
    );
    if (matches.length > 0) {
      score += Math.min(20, matches.length * 5);
      factors.push(`${matches.length} nyckelord matchar`);
    }
  }
  
  return {
    score: Math.min(100, score),
    factors
  };
}
```

```python
from playwright.sync_api import sync_playwright
from bs4 import BeautifulSoup
from datetime import datetime
import re
from utils.db import upsert_grant, log_scrape_result

class TillvaxtverketScraper:
    """
    Scraper for Tillväxtverket grant opportunities
    URL: https://tillvaxtverket.se/tillvaxtverket/sokfinansiering/utlysningar.html
    """
    
    def __init__(self, source_id):
        self.source_id = source_id
        self.base_url = "https://tillvaxtverket.se"
        self.list_url = "https://tillvaxtverket.se/tillvaxtverket/sokfinansiering/utlysningar.html"
    
    def scrape(self):
        print(f"Starting Tillväxtverket scrape from {self.list_url}")
        
        try:
            with sync_playwright() as p:
                browser = p.chromium.launch(headless=True)
                page = browser.new_page()
                
                # Navigate to grants page
                page.goto(self.list_url, wait_until='networkidle')
                
                # Wait for grant listings to load
                page.wait_for_selector('.sv-list-element', timeout=10000)
                
                # Get page content
                html = page.content()
                browser.close()
                
                # Parse grants
                grants = self.parse_grants(html)
                
                # Insert into database
                grants_inserted = 0
                for grant in grants:
                    if grant['title'] and grant['url']:
                        upsert_grant(grant)
                        grants_inserted += 1
                
                log_scrape_result(
                    source_id=self.source_id,
                    status='success',
                    grants_found=grants_inserted,
                    error_message=None
                )
                
                print(f"Successfully scraped {grants_inserted} grants from Tillväxtverket")
                return grants_inserted
                
        except Exception as e:
            error_msg = str(e)
            print(f"Tillväxtverket scrape failed: {error_msg}")
            
            log_scrape_result(
                source_id=self.source_id,
                status='failed',
                grants_found=0,
                error_message=error_msg
            )
            
            return 0
    
    def parse_grants(self, html):
        soup = BeautifulSoup(html, 'html.parser')
        grants = []
        
        # Find all grant list items
        grant_elements = soup.select('.sv-list-element')
        
        for element in grant_elements:
            try:
                # Extract title and link
                title_elem = element.select_one('h2 a, h3 a, .sv-list-element__title a')
                if not title_elem:
                    continue
                
                title = title_elem.get_text(strip=True)
                url = title_elem.get('href', '')
                
                # Make URL absolute
                if url and not url.startswith('http'):
                    url = self.base_url + url
                
                # Extract description
                desc_elem = element.select_one('.sv-list-element__description, .preamble')
                description = desc_elem.get_text(strip=True) if desc_elem else ''
                
                # Extract deadline if present
                deadline = None
                deadline_elem = element.select_one('.date, .deadline, time')
                if deadline_elem:
                    deadline_text = deadline_elem.get_text(strip=True)
                    deadline = self.parse_deadline(deadline_text)
                
                # Determine status
                status = 'open'
                if deadline and deadline < datetime.now():
                    status = 'closed'
                
                # Build grant object
                grant = {
                    'title': title,
                    'description': description,
                    'source_name': 'Tillväxtverket',
                    'source_type': 'myndighet',
                    'url': url,
                    'deadline': deadline,
                    'amount_min': None,
                    'amount_max': None,
                    'eligibility_criteria': {},
                    'target_group': ['sme', 'all'],
                    'keywords': self.extract_keywords(title, description),
                    'application_requirements': {},
                    'status': status,
                    'raw_data': {}
                }
                
                grants.append(grant)
                
            except Exception as e:
                print(f"Error parsing grant element: {e}")
                continue
        
        return grants
    
    def parse_deadline(self, date_text):
        """Parse Swedish date formats"""
        try:
            # Common formats:
            # "2026-03-15"
            # "15 mars 2026"
            # "15 mars"
            
            # Try ISO format first
            if re.match(r'\d{4}-\d{2}-\d{2}', date_text):
                return datetime.fromisoformat(date_text)
            
            # Try Swedish month names
            months_sv = {
                'januari': 1, 'februari': 2, 'mars': 3, 'april': 4,
                'maj': 5, 'juni': 6, 'juli': 7, 'augusti': 8,
                'september': 9, 'oktober': 10, 'november': 11, 'december': 12
            }
            
            for month_name, month_num in months_sv.items():
                if month_name in date_text.lower():
                    # Extract day and year
                    day_match = re.search(r'\d{1,2}', date_text)
                    year_match = re.search(r'\d{4}', date_text)
                    
                    if day_match:
                        day = int(day_match.group())
                        year = int(year_match.group()) if year_match else datetime.now().year
                        
                        return datetime(year, month_num, day)
            
        except Exception as e:
            print(f"Could not parse deadline: {date_text} - {e}")
        
        return None
    
    def extract_keywords(self, title, description):
        """Extract relevant keywords from text"""
        keywords = []
        text = f"{title} {description}".lower()
        
        # Common keywords to extract
        keyword_patterns = [
            'innovation', 'forskning', 'utveckling', 'hållbarhet',
            'export', 'internationalisering', 'digitalisering',
            'klimat', 'miljö', 'grön omställning', 'cirkulär ekonomi'
        ]
        
        for keyword in keyword_patterns:
            if keyword in text:
                keywords.append(keyword)
        
        return keywords[:10]  # Max 10 keywords
```

Create app/api/cron/notifications/route.ts:
```typescript
import { createClient } from '@/lib/supabase/server';
import { sendNewGrantNotification, sendDeadlineReminder } from '@/lib/email/notifier';

export async function GET(request: Request) {
  // Verify cron secret to prevent unauthorized access
  const authHeader = request.headers.get('authorization');
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 });
  }
  
  const supabase = createClient();
  
  try {
    // 1. Find new grants added in last 24 hours
    const yesterday = new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString();
    
    const { data: newGrants } = await supabase
      .from('grants')
      .select('*')
      .gte('created_at', yesterday)
      .eq('status', 'open');
    
    // 2. Get all companies with users
    const { data: companies } = await supabase
      .from('companies')
      .select(`
        *,
        user:users(email)
      `);
    
    // 3. Send new grant notifications
    let notificationsSent = 0;
    
    for (const grant of newGrants || []) {
      for (const company of companies || []) {
        // Calculate match score
        const matchScore = await calculateMatchScore(company, grant);
        
        // Only notify if match is >60%
        if (matchScore >= 60 && company.user?.email) {
          await sendNewGrantNotification(
            company.user.email,
            grant,
            company.company_name,
            matchScore
          );
          notificationsSent++;
        }
      }
    }
    
    // 4. Check for upcoming deadlines (7 days, 3 days, 1 day)
    const deadlineRanges = [7, 3, 1];
    
    for (const daysAhead of deadlineRanges) {
      const targetDate = new Date();
      targetDate.setDate(targetDate.getDate() + daysAhead);
      
      const { data: upcomingGrants } = await supabase
        .from('grants')
        .select('*')
        .eq('status', 'open')
        .gte('deadline', targetDate.toISOString().split('T')[0])
        .lte('deadline', new Date(targetDate.getTime() + 24*60*60*1000).toISOString().split('T')[0]);
      
      for (const grant of upcomingGrants || []) {
        for (const company of companies || []) {
          if (company.user?.email) {
            await sendDeadlineReminder(
              company.user.email,
              grant,
              company.company_name
            );
            notificationsSent++;
          }
        }
      }
    }
    
    return Response.json({
      success: true,
      notificationsSent,
      newGrantsFound: newGrants?.length || 0
    });
    
  } catch (error) {
    console.error('Notification cron error:', error);
    return Response.json({
      success: false,
      error: error.message
    }, { status: 500 });
  }
}
```

```typescript
import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';

const redis = new Redis({
  url: process.env.UPSTASH_REDIS_REST_URL!,
  token: process.env.UPSTASH_REDIS_REST_TOKEN!,
});

// Create different rate limiters
export const aiGenerationLimiter = new Ratelimit({
  redis,
  limiter: Ratelimit.slidingWindow(10, '1 h'), // 10 generations per hour
  analytics: true,
});

export const apiLimiter = new Ratelimit({
  redis,
  limiter: Ratelimit.slidingWindow(100, '1 m'), // 100 requests per minute
  analytics: true,
});

export async function checkRateLimit(identifier: string, limiter: Ratelimit) {
  const { success, limit, reset, remaining } = await limiter.limit(identifier);
  
  return {
    success,
    limit,
    remaining,
    reset: new Date(reset)
  };
}
```
