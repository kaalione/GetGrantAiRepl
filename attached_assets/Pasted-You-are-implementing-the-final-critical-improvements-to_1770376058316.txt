```typescript
import { db } from '../db';
import { userProgress } from '../db/schema';
import { eq } from 'drizzle-orm';

export async function getOrCreateUserProgress(userId: string) {
  let progress = await db.query.userProgress.findFirst({
    where: eq(userProgress.userId, userId),
  });

  if (!progress) {
    [progress] = await db.insert(userProgress)
      .values({ userId })
      .returning();
  }

  return progress;
}

export async function updateProgressMilestone(
  userId: string,
  milestone: keyof typeof userProgress.$inferInsert,
  value: boolean = true
) {
  const progress = await getOrCreateUserProgress(userId);

  // Only update if not already completed
  if (!progress[milestone]) {
    const updates: any = {
      [milestone]: value,
      updatedAt: new Date(),
    };

    // Also set timestamp field if it exists
    const timestampField = `${milestone}At` as keyof typeof userProgress.$inferInsert;
    if (timestampField in progress) {
      updates[timestampField] = new Date();
    }

    await db.update(userProgress)
      .set(updates)
      .where(eq(userProgress.userId, userId));
  }
}

export async function checkAndUpdateProgress(userId: string) {
  // Get user's company
  const company = await db.query.companies.findFirst({
    where: eq(companies.userId, userId),
  });

  if (company) {
    await updateProgressMilestone(userId, 'profileCreated', true);

    // Check if profile is >90% complete
    const completion = calculateProfileCompletion(company);
    if (completion.percentage >= 90) {
      await updateProgressMilestone(userId, 'profileCompleted', true);
    }
  }

  // Check for first grant view (from analytics or separate tracking)
  const hasViewedGrant = await checkIfUserViewedGrant(userId);
  if (hasViewedGrant) {
    await updateProgressMilestone(userId, 'firstGrantViewed', true);
  }

  // Check for first AI analysis
  const hasRunAnalysis = await db.query.grantMatches.findFirst({
    where: eq(grantMatches.userId, userId),
  });
  if (hasRunAnalysis) {
    await updateProgressMilestone(userId, 'firstAIAnalysisRun', true);
  }

  // Check for first application
  const hasApplication = await db.query.applications.findFirst({
    where: eq(applications.userId, userId),
  });
  if (hasApplication) {
    await updateProgressMilestone(userId, 'firstApplicationGenerated', true);
  }

  // Check for first submitted application
  const hasSubmitted = await db.query.applications.findFirst({
    where: and(
      eq(applications.userId, userId),
      eq(applications.status, 'submitted')
    ),
  });
  if (hasSubmitted) {
    await updateProgressMilestone(userId, 'firstApplicationSubmitted', true);
  }
}

async function checkIfUserViewedGrant(userId: string): Promise<boolean> {
  // Check if user has any grant views in analytics or tracking table

  // For now, return true if they have any applications
  const hasApp = await db.query.applications.findFirst({
    where: eq(applications.userId, userId),
  });
  return !!hasApp;
}
```

STEP 2.1: Create Deadline Utility Functions

Create `src/lib/deadlines.ts`:
```typescript
import { differenceInDays, isBefore, startOfDay } from 'date-fns';

export type DeadlineUrgency = 'critical' | 'urgent' | 'upcoming' | 'normal';

export interface CategorizedDeadlines {
  critical: any[]; // <3 days
  urgent: any[];   // 3-7 days
  upcoming: any[]; // 7-14 days
}

export function getDeadlineUrgency(deadline: Date): DeadlineUrgency {
  const now = startOfDay(new Date());
  const deadlineDate = startOfDay(new Date(deadline));
  const daysUntil = differenceInDays(deadlineDate, now);

  if (daysUntil < 0) return 'normal'; // Past deadline
  if (daysUntil <= 3) return 'critical';
  if (daysUntil <= 7) return 'urgent';
  if (daysUntil <= 14) return 'upcoming';
  return 'normal';
}

export function categorizeGrantsByDeadline(grants: any[]): CategorizedDeadlines {
  const result: CategorizedDeadlines = {
    critical: [],
    urgent: [],
    upcoming: [],
  };

  grants.forEach(grant => {
    if (!grant.deadline) return;
    
    const urgency = getDeadlineUrgency(grant.deadline);
    
    if (urgency === 'critical') result.critical.push(grant);
    else if (urgency === 'urgent') result.urgent.push(grant);
    else if (urgency === 'upcoming') result.upcoming.push(grant);
  });

  return result;
}

export function getUrgencyColor(urgency: DeadlineUrgency) {
  switch (urgency) {
    case 'critical':
      return {
        bg: 'bg-red-50 dark:bg-red-900/20',
        border: 'border-red-500',
        text: 'text-red-900 dark:text-red-100',
        icon: 'text-red-600',
      };
    case 'urgent':
      return {
        bg: 'bg-amber-50 dark:bg-amber-900/20',
        border: 'border-amber-500',
        text: 'text-amber-900 dark:text-amber-100',
        icon: 'text-amber-600',
      };
    case 'upcoming':
      return {
        bg: 'bg-blue-50 dark:bg-blue-900/20',
        border: 'border-blue-500',
        text: 'text-blue-900 dark:text-blue-100',
        icon: 'text-blue-600',
      };
    default:
      return {
        bg: 'bg-gray-50 dark:bg-gray-800',
        border: 'border-gray-300',
        text: 'text-gray-900 dark:text-gray-100',
        icon: 'text-gray-600',
      };
  }
}

export function getUrgencyLabel(urgency: DeadlineUrgency, t: any) {
  switch (urgency) {
    case 'critical': return t('deadlines.critical');
    case 'urgent': return t('deadlines.urgent');
    case 'upcoming': return t('deadlines.upcoming');
    default: return '';
  }
}

export function getUrgencyDescription(urgency: DeadlineUrgency, t: any) {
  switch (urgency) {
    case 'critical': return t('deadlines.criticalDesc');
    case 'urgent': return t('deadlines.urgentDesc');
    case 'upcoming': return t('deadlines.upcomingDesc');
    default: return '';
  }
}
```

```typescript
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { AlertTriangle, Clock, Calendar } from 'lucide-react';

import { sv } from 'date-fns/locale';
import { useTranslation } from 'react-i18next';
import { categorizeGrantsByDeadline, getUrgencyColor, getUrgencyLabel, getUrgencyDescription } from '@/lib/deadlines';

interface DeadlineAlertsProps {
  grants: any[];
}

export function DeadlineAlerts({ grants }: DeadlineAlertsProps) {
  const { t, i18n } = useTranslation();
  const locale = i18n.language === 'sv' ? sv : undefined;

  const categorized = categorizeGrantsByDeadline(grants);

      icon: Calendar,
      grants: categorized.upcoming,
```

```typescript
      where: and(
        eq(grants.status, 'open'),
        gte(grants.deadline, now),
        lte(grants.deadline, twoWeeksFromNow)
      ),
      orderBy: [asc(grants.deadline)],
    });

    // Get user's company for match scores
    const company = await db.query.companies.findFirst({
      where: eq(companies.userId, req.user.id),
    });

    // Add match scores if company exists
    const grantsWithScores = grants.map(grant => {
      if (company) {
        const match = calculateBasicMatch(company, grant);
        return {
          ...grant,
          matchScore: match.score,
        };
      }
      return grant;
    });

    res.json(grantsWithScores);
  } catch (error) {
    console.error('Fetch upcoming deadlines error:', error);
    res.status(500).json({ error: 'Failed to fetch grants' });
  }
});
```
